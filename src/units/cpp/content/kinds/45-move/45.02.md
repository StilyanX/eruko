# 45.02
Source: written
Write a class `ship::Ship` whose objects stand on a grid of whole-number points, each facing one of the values of an enum class `ship::Compass`: `NORTH`, `EAST`, `SOUTH` and `WEST`.
East grows to the east and north grows to the north.
`Ship(int east, int north, Compass compass)` stands at the point (east, north) facing compass.
`east()`, `north()` and `compass()` return where it stands and the way it faces.
`starboard()` turns it a quarter turn to the right and `port()` a quarter turn to the left, in place.
`sail()` moves it 1 point the way it faces.
`follow(const std::string& orders)` runs each letter of the text in order: `S` turns to the right, `P` turns to the left and `F` sails.
Limits: -10 ≤ east, north ≤ 10 at the start; 0 ≤ length of the orders ≤ 20, each `S`, `P` or `F`.
## Solution
```cpp
#pragma once

#include <string>

namespace ship {

//: Name each possible value in an `enum class`.
enum class Compass { [[NORTH]], EAST, SOUTH, WEST };

//: Write a class that keeps its data in private members.
class Ship {
    int east_ [[{0}]];
    int north_ {0};
    Compass compass_ {Compass::NORTH};
public:
    //: Write a constructor that sets the data.
    Ship(int east, int north, Compass compass)
        : east_ {[[east]]}, north_ {north}, compass_ {compass}
    {
    }
    //: Write member functions that give back the data.
    int east() const
    {
        return [[east_]];
    }
    int north() const
    {
        return north_;
    }
    Compass compass() const
    {
        return compass_;
    }
    //: Write member functions that turn it a quarter turn to the right or to the left.
    void starboard()
    {
        const int next {[[(static_cast<int>(compass_) + 1) % 4]]};
        compass_ = static_cast<Compass>(next);
    }
    void port()
    {
        const int next {(static_cast<int>(compass_) + 3) % 4};
        compass_ = static_cast<Compass>(next);
    }
    //: Write a member function that moves it one point in the direction it faces.
    void sail()
    {
        if ([[compass_ == Compass::NORTH]]) {
            ++north_;
        } else if (compass_ == Compass::EAST) {
            ++east_;
        } else if (compass_ == Compass::SOUTH) {
            --north_;
        } else {
            --east_;
        }
    }
    //: Write a member function that runs each instruction of the text in turn.
    void follow(const std::string& orders)
    {
        for (const char order : orders) {
            if ([[order == 'S']]) {
                starboard();
            } else if (order == 'P') {
                port();
            } else {
                sail();
            }
        }
    }
};

}  // namespace ship
```
## Slow
```cpp
#pragma once

#include <string>

namespace ship {

enum class Compass { NORTH, EAST, SOUTH, WEST };

class Ship {
    int east_ {0};
    int north_ {0};
    Compass compass_ {Compass::NORTH};
public:
    Ship(int east, int north, Compass compass)
        : east_ {east}, north_ {north}, compass_ {compass}
    {
    }
    int east() const
    {
        return east_;
    }
    int north() const
    {
        return north_;
    }
    Compass compass() const
    {
        return compass_;
    }
    void starboard()
    {
        if (compass_ == Compass::NORTH) {
            compass_ = Compass::EAST;
        } else if (compass_ == Compass::EAST) {
            compass_ = Compass::SOUTH;
        } else if (compass_ == Compass::SOUTH) {
            compass_ = Compass::WEST;
        } else {
            compass_ = Compass::NORTH;
        }
    }
    void port()
    {
        starboard();
        starboard();
        starboard();
    }
    void sail()
    {
        const int east_steps[] {0, 1, 0, -1};
        const int north_steps[] {1, 0, -1, 0};
        east_ += east_steps[static_cast<int>(compass_)];
        north_ += north_steps[static_cast<int>(compass_)];
    }
    void follow(const std::string& orders)
    {
        for (const char order : orders) {
            if (order == 'F') {
                sail();
            } else if (order == 'P') {
                port();
            } else {
                starboard();
            }
        }
    }
};

}  // namespace ship
```
## Tests
```cpp
TEST_CASE("sail_twice_after_turning_right") {
    ship::Ship boat {0, 0, ship::Compass::NORTH};
    boat.follow("SFF");
    REQUIRE(boat.east() == 2);
    REQUIRE(boat.north() == 0);
    REQUIRE(boat.compass() == ship::Compass::EAST);
}
TEST_CASE("start_is_kept") {
    const ship::Ship boat {-10, 10, ship::Compass::WEST};
    REQUIRE(boat.east() == -10);
    REQUIRE(boat.north() == 10);
    REQUIRE(boat.compass() == ship::Compass::WEST);
}
TEST_CASE("starboard_turns_from_each_way") {
    ship::Ship boat {0, 0, ship::Compass::NORTH};
    boat.starboard();
    REQUIRE(boat.compass() == ship::Compass::EAST);
    boat.starboard();
    REQUIRE(boat.compass() == ship::Compass::SOUTH);
    boat.starboard();
    REQUIRE(boat.compass() == ship::Compass::WEST);
    boat.starboard();
    REQUIRE(boat.compass() == ship::Compass::NORTH);
}
TEST_CASE("port_turns_from_each_way") {
    ship::Ship boat {0, 0, ship::Compass::NORTH};
    boat.port();
    REQUIRE(boat.compass() == ship::Compass::WEST);
    boat.port();
    REQUIRE(boat.compass() == ship::Compass::SOUTH);
    boat.port();
    REQUIRE(boat.compass() == ship::Compass::EAST);
    boat.port();
    REQUIRE(boat.compass() == ship::Compass::NORTH);
}
TEST_CASE("sail_each_way") {
    ship::Ship north {1, 1, ship::Compass::NORTH};
    north.sail();
    REQUIRE(north.north() == 2);
    ship::Ship east {1, 1, ship::Compass::EAST};
    east.sail();
    REQUIRE(east.east() == 2);
    ship::Ship south {1, 1, ship::Compass::SOUTH};
    south.sail();
    REQUIRE(south.north() == 0);
    ship::Ship west {1, 1, ship::Compass::WEST};
    west.sail();
    REQUIRE(west.east() == 0);
}
TEST_CASE("no_orders") {
    ship::Ship boat {3, -4, ship::Compass::SOUTH};
    boat.follow("");
    REQUIRE(boat.east() == 3);
    REQUIRE(boat.north() == -4);
    REQUIRE(boat.compass() == ship::Compass::SOUTH);
}
TEST_CASE("long_orders") {
    ship::Ship boat {7, 3, ship::Compass::NORTH};
    boat.follow("PFFFFFSSFFPFPFFFFSFF");
    REQUIRE(boat.east() == 0);
    REQUIRE(boat.north() == 6);
    REQUIRE(boat.compass() == ship::Compass::NORTH);
}
```
