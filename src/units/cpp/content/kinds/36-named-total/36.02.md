# 36.02
Source: written
Write a function `steps::place(const std::string& moves)` that returns where a walk ends, from its moves up and down.
Each `U` in moves is one step up and each `D` one step down, from level 0.
The header also defines the result type, `enum class steps::level { below, ground, above }`.
It returns `below` when the walk ends below level 0, `ground` when it ends at level 0, and `above` when it ends above it.
Limits: 0 ≤ length of moves ≤ 100; each character of moves is `U` or `D`.
## Solution
```cpp
#pragma once

#include <string>

namespace steps {

//: Name each possible value in an `enum class`.
enum class level { below, [[ground]], above };

//: Write a function that adds to a total, then gives back an answer from the total.
level place([[const std::string& moves]])
{
    //:: Start the total at zero.
    int height {[[0]]};
    //:: Add 1 for a `U` and take 1 for a `D`.
    for (const char move : moves) {
        if ([[move == 'U']]) {
            ++height;
        } else {
            --height;
        }
    }
    //:: Give back below for a total below 0, ground for 0, else above.
    if ([[height < 0]]) {
        return level::below;
    } else if (height == 0) {
        return level::ground;
    } else {
        return level::above;
    }
}

}  // namespace steps
```
## Slow
```cpp
#pragma once

#include <string>

namespace steps {

enum class level { below, ground, above };

level place(const std::string& moves)
{
    int ups {0};
    int downs {0};
    for (std::size_t index {0}; index < moves.size(); ++index) {
        if (moves[index] == 'U') {
            ++ups;
        }
        if (moves[index] == 'D') {
            ++downs;
        }
    }
    if (ups > downs) {
        return level::above;
    }
    if (ups < downs) {
        return level::below;
    }
    return level::ground;
}

}  // namespace steps
```
## Tests
```cpp
TEST_CASE("two_up_one_down") { REQUIRE(steps::place("UDU") == steps::level::above); }
TEST_CASE("no_moves") { REQUIRE(steps::place("") == steps::level::ground); }
TEST_CASE("one_up") { REQUIRE(steps::place("U") == steps::level::above); }
TEST_CASE("one_down") { REQUIRE(steps::place("D") == steps::level::below); }
TEST_CASE("back_to_ground") { REQUIRE(steps::place("UUDD") == steps::level::ground); }
TEST_CASE("down_then_up") { REQUIRE(steps::place("DDDUU") == steps::level::below); }
TEST_CASE("hundred_ups") { REQUIRE(steps::place(std::string(100, 'U')) == steps::level::above); }
TEST_CASE("hundred_downs") { REQUIRE(steps::place(std::string(100, 'D')) == steps::level::below); }
TEST_CASE("fifty_each") { REQUIRE(steps::place(std::string(50, 'D') + std::string(50, 'U')) == steps::level::ground); }
```
