# 32.02
Source: written
Write a function `case_code::encode(const std::string& text)` that returns the text with each small letter turned into its capital, each capital turned into its small letter, and each space turned into `-`.
Digits stay as they are.
Write a function `case_code::decode(const std::string& text)` that returns the text that `encode` turns into the given text.
Empty text gives empty text.
Limits: 0 ≤ length of text ≤ 1000; text holds letters, digits and spaces only.
## Solution
```cpp
#pragma once

#include <cctype>
#include <string>

namespace case_code {

//: Write a function that tests each case in order and gives back the result of the first that holds.
char swap([[char character]])
{
    //:: Give back a small letter as its capital.
    if ([[std::islower(character)]]) {
        return static_cast<char>(std::toupper(character));
    }
    //:: Give back a capital as its small letter, else the character as it is.
    return static_cast<char>([[std::tolower(character)]]);
}

//: Write a function that builds a new string from each character of the text.
std::string encode([[const std::string& text]])
{
    //:: Start the result as an empty string.
    std::string result {[[""]]};
    //:: Add `-` for a space, else the character swapped.
    for (const char character : text) {
        if ([[character == ' ']]) {
            result += '-';
        } else {
            result += swap(character);
        }
    }
    //:: Give back the result.
    return [[result]];
}

//: Write a function that builds a new string from each character of the text.
std::string decode([[const std::string& text]])
{
    //:: Start the result as an empty string.
    std::string result {[[""]]};
    //:: Add a space for `-`, else the character swapped.
    for (const char character : text) {
        if ([[character == '-']]) {
            result += ' ';
        } else {
            result += swap(character);
        }
    }
    //:: Give back the result.
    return [[result]];
}

}  // namespace case_code
```
## Slow
```cpp
#pragma once

#include <string>

namespace case_code {

const std::string small_letters {"abcdefghijklmnopqrstuvwxyz"};
const std::string capitals {"ABCDEFGHIJKLMNOPQRSTUVWXYZ"};

std::string turn(const std::string& text, char from, char to)
{
    std::string result {text};
    for (std::size_t i {0}; i < result.size(); ++i) {
        bool done {false};
        for (std::size_t k {0}; k < 26 && !done; ++k) {
            if (text[i] == small_letters[k]) {
                result[i] = capitals[k];
                done = true;
            } else if (text[i] == capitals[k]) {
                result[i] = small_letters[k];
                done = true;
            }
        }
        if (!done && text[i] == from) {
            result[i] = to;
        }
    }
    return result;
}

std::string encode(const std::string& text)
{
    return turn(text, ' ', '-');
}

std::string decode(const std::string& text)
{
    return turn(text, '-', ' ');
}

}  // namespace case_code
```

TEST_CASE("encode_words") {
    REQUIRE(case_code::encode("Hello World 2") == "hELLO-wORLD-2");

TEST_CASE("encode_empty") { REQUIRE(case_code::encode("") == ""); }
TEST_CASE("decode_empty") { REQUIRE(case_code::decode("") == ""); }
TEST_CASE("encode_one_small") { REQUIRE(case_code::encode("a") == "A"); }
TEST_CASE("encode_one_capital") {
    REQUIRE(case_code::encode("Z") == "z");
}
TEST_CASE("encode_one_space") { REQUIRE(case_code::encode(" ") == "-"); }
TEST_CASE("encode_digits") {
    REQUIRE(case_code::encode("0123456789") == "0123456789");
}
TEST_CASE("encode_all_letters") {
    REQUIRE(case_code::encode("abcxyzABCXYZ") == "ABCXYZabcxyz");
}
TEST_CASE("encode_spaces") {
    REQUIRE(case_code::encode("a  b ") == "A--B-");
}

    REQUIRE(case_code::decode("hELLO-wORLD-2") == "Hello World 2");
}

    REQUIRE(case_code::decode("42") == "42");

TEST_CASE("round_trip_largest") {
    std::string text {};
    for (int i {0}; i < 250; ++i) {
        text += "aB 7";
    }
    REQUIRE(case_code::decode(case_code::encode(text)) == text);
    REQUIRE(case_code::encode(text).size() == 1000);
}
