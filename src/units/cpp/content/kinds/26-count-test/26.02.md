# 26.02
Source: written
Write a function `digits::count_digits(const std::string& text)` that returns how many characters of the text are digits from `0` to `9`.
An empty text gives 0.
Limits: 0 ≤ length of the text ≤ 1000.
## Solution
```cpp
#pragma once

#include <cctype>
#include <string>

namespace digits {

//: Write a function that counts the items that pass a test and gives back the count.
int count_digits([[const std::string& text]])
{
    //:: Start the count at zero.
    int count {[[0]]};
    //:: Add one for each digit.
    for (const char character : text) {
        if ([[std::isdigit(character)]]) {
            ++count;
        }
    }
    //:: Give back the count.
    return [[count]];
}

}  // namespace digits
```
## Slow
```cpp
#pragma once

#include <string>

namespace digits {

int count_digits(const std::string& text)
{
    int found {0};
    for (char digit {'0'}; digit <= '9'; ++digit) {
        for (std::size_t i {0}; i < text.size(); ++i) {
            if (text[i] == digit) {
                found = found + 1;
            }
        }
    }
    return found;
}

}  // namespace digits
```
## Tests
```cpp
TEST_CASE("mixed") { REQUIRE(digits::count_digits("a1b22c") == 3); }
TEST_CASE("empty") { REQUIRE(digits::count_digits("") == 0); }
TEST_CASE("one_digit") { REQUIRE(digits::count_digits("7") == 1); }
TEST_CASE("one_letter") { REQUIRE(digits::count_digits("x") == 0); }
TEST_CASE("no_digits") {
    REQUIRE(digits::count_digits("No digits here!") == 0);
}
TEST_CASE("only_digits") {
    REQUIRE(digits::count_digits("0123456789") == 10);
}
TEST_CASE("spaces_and_signs") {
    REQUIRE(digits::count_digits("-4 + 5 = 1, (9)") == 4);
}
TEST_CASE("largest") {
    REQUIRE(digits::count_digits(std::string(1000, '9')) == 1000);
}
TEST_CASE("largest_mixed") {
    std::string text {};
    for (int i {0}; i < 500; ++i) {
        text += "a5";
    }
    REQUIRE(digits::count_digits(text) == 500);
}
```
