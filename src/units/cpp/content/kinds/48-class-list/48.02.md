# 48.02
Source: written
Write a class `weather::readings` made from a `std::vector<int>` of temperatures, in the order they were taken.
`all()` returns all the temperatures, in the same order.
`latest()` returns the temperature taken last.
`lowest()` returns the smallest temperature.
`highest_five()` returns the 5 largest temperatures, largest first, or all of them, largest first, when there are fewer than 5; equal temperatures each count.
No member function changes the temperatures.
Limits: 1 ≤ number of temperatures ≤ 1000; -100 ≤ each temperature ≤ 100.
## Solution
```cpp
#pragma once

#include <algorithm>
#include <functional>
#include <vector>

namespace weather {

//: Write a class with a constructor that sets the data.
class readings {
public:
    explicit readings([[std::vector<int> temperatures]])
        : temperatures_ {temperatures}
    {
    }
    //: Write a member function that gives back the data.
    std::vector<int> all() const
    {
        return [[temperatures_]];
    }
    //: Write a member function that gives back a value worked out in one step.
    int latest() const
    {
        return [[temperatures_.back()]];
    }
    //: Write a member function that gives back a value worked out in one step.
    int lowest() const
    {
        return [[std::ranges::min(temperatures_)]];
    }
    //: Write a member function that gives back the largest values of a list, largest first.
    [[std::vector<int>]] highest_five() const
    {
        //:: Copy the list, and sort it from largest to smallest.
        std::vector<int> sorted {[[temperatures_]]};
        std::ranges::sort(sorted, std::ranges::greater {});
        //:: Keep at most the first 5.
        sorted.resize([[std::min(sorted.size(), std::size_t {5})]]);
        //:: Give back the sorted list.
        return [[sorted]];
    }
//: Keep the data in private members.
private:
    [[std::vector<int>]] temperatures_ {};
};

}  // namespace weather
```
## Slow
```cpp
#pragma once

#include <vector>

namespace weather {

class readings {
public:
    explicit readings(std::vector<int> temperatures)
        : list_ {temperatures}
    {
    }
    std::vector<int> all() const
    {
        return list_;
    }
    int latest() const
    {
        return list_[list_.size() - 1];
    }
    int lowest() const
    {
        int low {list_[0]};
        for (const int t : list_) {
            if (t < low) {
                low = t;
            }
        }
        return low;
    }
    std::vector<int> highest_five() const
    {
        std::vector<int> left {list_};
        std::vector<int> result {};
        while (result.size() < 5 && !left.empty()) {
            std::size_t best {0};
            for (std::size_t i {1}; i < left.size(); ++i) {
                if (left[i] > left[best]) {
                    best = i;
                }
            }
            result.push_back(left[best]);
            left.erase(left.begin() + static_cast<long>(best));
        }
        return result;
    }
private:
    std::vector<int> list_ {};
};

}  // namespace weather
```
## Tests
```cpp
TEST_CASE("highest_five_of_seven") {
    const weather::readings r {{3, -1, 7, 7, 0, 12, 5}};
    REQUIRE(r.highest_five() == std::vector<int> {12, 7, 7, 5, 3});
}
TEST_CASE("all_in_order") {
    const weather::readings r {{3, -1, 7}};
    REQUIRE(r.all() == std::vector<int> {3, -1, 7});
}
TEST_CASE("latest_is_last") {
    const weather::readings r {{3, -1, 7, 2}};
    REQUIRE(r.latest() == 2);
}
TEST_CASE("lowest_of_several") {
    const weather::readings r {{3, -1, 7, -100, 100}};
    REQUIRE(r.lowest() == -100);
}
TEST_CASE("one_temperature") {
    const weather::readings r {{-5}};
    REQUIRE(r.all() == std::vector<int> {-5});
    REQUIRE(r.latest() == -5);
    REQUIRE(r.lowest() == -5);
    REQUIRE(r.highest_five() == std::vector<int> {-5});
}
TEST_CASE("fewer_than_five") {
    const weather::readings r {{1, 4, 2}};
    REQUIRE(r.highest_five() == std::vector<int> {4, 2, 1});
}
TEST_CASE("exactly_five") {
    const weather::readings r {{9, 8, 10, 6, 7}};
    REQUIRE(r.highest_five() == std::vector<int> {10, 9, 8, 7, 6});
}
TEST_CASE("all_equal") {
    const weather::readings r {{4, 4, 4, 4, 4, 4}};
    REQUIRE(r.highest_five() == std::vector<int> {4, 4, 4, 4, 4});
}
TEST_CASE("highest_five_leaves_the_list_alone") {
    const weather::readings r {{2, 9, 1}};
    REQUIRE(r.highest_five() == std::vector<int> {9, 2, 1});
    REQUIRE(r.all() == std::vector<int> {2, 9, 1});
}
```
