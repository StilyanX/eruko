# 37.02
Source: written
Write a function `calendar::days_before(int month)` that returns how many days of a year that is not a leap year come before the first day of the month.
The months 1 to 12 have 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30 and 31 days.
Month 1 gives 0.
Limits: 1 ≤ month ≤ 12.
## Solution
```cpp
#pragma once

#include <array>

namespace calendar {

//: Make a table of fixed values.
const [[std::array<int, 12>]] days {
    31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};

//: Write a function that adds up a sequence and gives back the total.
int days_before([[int month]])
{
    //:: Start the total at zero.
    int total {[[0]]};
    //:: Add the days of each month before the month.
    for (int index {0}; index < [[month - 1]]; ++index) {
        total += days.at(index);
    }
    //:: Give back the total.
    return [[total]];
}

}  // namespace calendar
```
## Slow
```cpp
#pragma once

namespace calendar {

int days_before(int month)
{
    int day {0};
    int current {1};
    int left {31};
    while (current < month) {
        ++day;
        --left;
        if (left == 0) {
            ++current;
            if (current == 2) {
                left = 28;
            } else if (current == 4 || current == 6 || current == 9
                       || current == 11) {
                left = 30;
            } else {
                left = 31;
            }
        }
    }
    return day;
}

}  // namespace calendar
```
