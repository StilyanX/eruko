# 25.02
Source: written
Write a class `inventory::store` that keeps an amount for each item, with these member functions.
`add(const std::string& item, int number)` adds number to the amount of the item.
`stock() const` returns a `std::map<std::string, int>` from each item added to its amount.
`amount(const std::string& item) const` returns the amount of the item, or 0 for an item never added.
A new store holds no items.
Limits: 1 ≤ number ≤ 1000; 0 ≤ items added ≤ 100; each item is 1 to 20 small letters.
## Solution
```cpp
#pragma once

#include <map>
#include <string>

namespace inventory {

//: Write a class that keeps its data in private members.
class store {
    [[std::map<std::string, int>]] amounts_ {};
public:
    //: Write a member function that adds an item to the data.
    void add(const std::string& item, int number)
    {
        amounts_[item] += [[number]];
    }
    //: Write a member function that gives back the data.
    std::map<std::string, int> stock() const
    {
        return [[amounts_]];
    }
    //: Write a member function that gives back part of the data.
    int amount([[const std::string& item]]) const
    {
        //:: Give back 0 when the map has no such item.
        if ([[!amounts_.contains(item)]]) {
            return 0;
        }
        //:: Give back the amount of the item.
        return [[amounts_.at(item)]];
    }
};

}  // namespace inventory
```
## Slow
```cpp
#pragma once

#include <map>
#include <string>
#include <utility>
#include <vector>

namespace inventory {

class store {
    std::vector<std::pair<std::string, int>> added_ {};
public:
    void add(const std::string& item, int number)
    {
        added_.push_back({item, number});
    }
    std::map<std::string, int> stock() const
    {
        std::map<std::string, int> result {};
        for (const auto& [item, number] : added_) {
            result[item] = amount(item);
        }
        return result;
    }
    int amount(const std::string& item) const
    {
        int total {0};
        for (const auto& [name, number] : added_) {
            if (name == item) {
                total += number;
            }
        }
        return total;
    }
};

}  // namespace inventory
```
## Tests
```cpp
TEST_CASE("amounts_add_up")
{
    inventory::store shop {};
    shop.add("apple", 3);
    shop.add("pear", 2);
    shop.add("apple", 4);
    REQUIRE(shop.amount("apple") == 7);
}
TEST_CASE("new_store_has_no_items")
{
    const inventory::store shop {};
    REQUIRE(shop.stock().empty());
}
TEST_CASE("item_never_added_has_amount_0")
{
    inventory::store shop {};
    shop.add("pear", 5);
    REQUIRE(shop.amount("plum") == 0);
}
TEST_CASE("one_item_added_once")
{
    inventory::store shop {};
    shop.add("fig", 1);
    REQUIRE(shop.amount("fig") == 1);
    const std::map<std::string, int> expected {{"fig", 1}};
    REQUIRE(shop.stock() == expected);
}
TEST_CASE("stock_holds_every_item")
{
    inventory::store shop {};
    shop.add("pear", 2);
    shop.add("apple", 3);
    shop.add("pear", 10);
    const std::map<std::string, int> expected {
        {"apple", 3}, {"pear", 12}};
    REQUIRE(shop.stock() == expected);
}
TEST_CASE("largest_amounts")
{
    inventory::store shop {};
    for (int index {0}; index < 100; ++index) {
        shop.add("plum", 1000);
    }
    REQUIRE(shop.amount("plum") == 100000);
}
TEST_CASE("many_items")
{
    inventory::store shop {};
    for (int index {0}; index < 100; ++index) {
        shop.add(std::string(1 + index % 20, 'a'), index + 1);
    }
    REQUIRE(shop.stock().size() == 20);
    REQUIRE(shop.amount("a") == 1 + 21 + 41 + 61 + 81);
}
```
