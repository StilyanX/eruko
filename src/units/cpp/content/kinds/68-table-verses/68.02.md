# 68.02
Source: written
Write a function `fruit_song::verse(unsigned number)` that returns verse number 1 to 4 of a song as a `std::string`, each line ending in `\n`.
The 4 fruits, in order, are apple, banana, pear and plum, and their colours are red, yellow, green and purple.
Verse n opens `Basket <n> holds:`.
Then, for each fruit from the first to fruit n, comes `- a <colour> <fruit>`.
Verse 4 then ends `The basket is full.`, and each other verse `There is room for more.`
Also write `fruit_song::verses(unsigned first, unsigned last)`, which returns the verses from first to last, each followed by an empty line.
Also write `fruit_song::sing()`, which returns `verses(1, 4)`.
Limits: 1 ≤ number ≤ 4; 1 ≤ first ≤ last ≤ 4.
## Solution
```cpp
#pragma once

#include <array>
#include <string>

namespace fruit_song {

//: Make a table of fixed values.
const std::array<std::string, 4> fruits {
    [["apple"]], "banana", "pear", "plum"};

//: Make a table of fixed values.
const std::array<std::string, 4> colours {
    [["red"]], "yellow", "green", "purple"};

//: Write a function that builds a new string one piece at a time and gives it back.
std::string verse([[unsigned number]])
{
    //:: Start the verse with the line that names its number.
    std::string result {"Basket " + [[std::to_string(number)]]};
    result += " holds:\n";
    //:: Add a line for each fruit from the first to this one, with its colour.
    for (unsigned index {0}; [[index < number]]; ++index) {
        result += "- a " + colours[index] + " " + fruits[index]
                  + "\n";
    }
    //:: Add the last line and give back the verse.
    if ([[number == 4]]) {
        result += "The basket is full.\n";
    } else {
        result += "There is room for more.\n";
    }
    return result;
}

//: Write a function that builds a new string one piece at a time and gives it back.
std::string verses([[unsigned first]], unsigned last)
{
    //:: Start the result as an empty string.
    std::string result [[{}]];
    //:: Add each verse, then an empty line.
    for (unsigned number {first}; number <= last; ++number) {
        result += [[verse(number)]] + "\n";
    }
    //:: Give back the result.
    return [[result]];
}

//: Write a function that calls the functions above and gives back the answer.
[[std::string]] sing()
{
    //:: Give back the verses from the first to the last.
    return [[verses(1, 4)]];
}

}  // namespace fruit_song
```
## Slow
```cpp
#pragma once

#include <string>

namespace fruit_song {

std::string item(unsigned index)
{
    if (index == 1) {
        return "- a red apple\n";
    }
    if (index == 2) {
        return "- a yellow banana\n";
    }
    if (index == 3) {
        return "- a green pear\n";
    }
    return "- a purple plum\n";
}

std::string verse(unsigned number)
{
    std::string text {"Basket "};
    text += static_cast<char>('0' + number);
    text += " holds:\n";
    for (unsigned i {1}; i <= number; ++i) {
        text += item(i);
    }
    text += number == 4 ? "The basket is full.\n"
                        : "There is room for more.\n";
    return text;
}

std::string verses(unsigned first, unsigned last)
{
    std::string text {};
    unsigned number {first};
    while (number <= last) {
        text += verse(number);
        text += "\n";
        number = number + 1;
    }
    return text;
}

std::string sing()
{
    return verse(1) + "\n" + verse(2) + "\n" + verse(3) + "\n"
           + verse(4) + "\n";
}

}  // namespace fruit_song
```
