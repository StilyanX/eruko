# 72.02
Source: written
Write a class `turnstile::gate` with the members `unlock()`, `pass()` and `passes()`, safe to call from many threads at once.
A new gate is locked and has let 0 people through.
`unlock()` unlocks it; `pass()` lets one person through, adds one to the number of passes, and locks the gate again; `passes()` returns the number of passes.
Each throws `std::runtime_error`: `unlock()` when the gate is unlocked; `pass()` when it is locked; and `passes()` when it is unlocked.
Limits: none
## Solution
```cpp
#pragma once

#include <mutex>
#include <stdexcept>

namespace turnstile {

//: Write a class that keeps its data in private members.
class gate {
    int passes_ [[{0}]];
    bool unlocked_ {false};
    std::mutex mutex_ {};
public:
    //: Write member functions that lock the data, stop with an error when a test fails, else change it.
    [[void]] unlock()
    {
        //:: Stop with an error when the gate is unlocked, else unlock it.
        const std::lock_guard lock {mutex_};
        if ([[unlocked_]]) {
            throw std::runtime_error {"the gate is unlocked"};
        }
        unlocked_ = true;
    }
    void pass()
    {
        //:: Stop with an error when the gate is locked, else add one pass and lock it.
        const std::lock_guard lock {[[mutex_]]};
        if (!unlocked_) {
            throw std::runtime_error {"the gate is locked"};
        }
        ++passes_;
        unlocked_ = false;
    }
    //: Write a member function that locks the data, stops with an error when a test fails, else gives back the data.
    int passes()
    {
        const std::lock_guard lock {mutex_};
        if ([[unlocked_]]) {
            throw std::runtime_error {"the gate is unlocked"};
        }
        return passes_;
    }
};

}  // namespace turnstile
```
## Slow
```cpp
#pragma once

#include <mutex>
#include <stdexcept>
#include <vector>

namespace turnstile {

class gate {
public:
    void unlock()
    {
        mutex_.lock();
        const bool was {unlocked_};
        unlocked_ = true;
        mutex_.unlock();
        if (was) {
            throw std::runtime_error {"the gate is unlocked"};
        }
    }
    void pass()
    {
        mutex_.lock();
        const bool was {unlocked_};
        if (was) {
            log_.push_back(true);
            unlocked_ = false;
        }
        mutex_.unlock();
        if (!was) {
            throw std::runtime_error {"the gate is locked"};
        }
    }
    int passes()
    {
        mutex_.lock();
        const bool was {unlocked_};
        int count {0};
        for (std::size_t i {0}; i < log_.size(); ++i) {
            ++count;
        }
        mutex_.unlock();
        if (was) {
            throw std::runtime_error {"the gate is unlocked"};
        }
        return count;
    }
private:
    std::vector<bool> log_ {};
    bool unlocked_ {false};
    std::mutex mutex_ {};
};

}  // namespace turnstile
```
## Tests
```cpp
TEST_CASE("two_people_through") {
    turnstile::gate g {};
    g.unlock();
    g.pass();
    g.unlock();
    g.pass();
    REQUIRE(g.passes() == 2);
}
TEST_CASE("new_gate_has_no_passes") {
    turnstile::gate g {};
    REQUIRE(g.passes() == 0);
}
TEST_CASE("pass_a_locked_gate") {
    turnstile::gate g {};
    REQUIRE_THROWS_AS(g.pass(), std::runtime_error);
}
TEST_CASE("unlock_twice") {
    turnstile::gate g {};
    g.unlock();
    REQUIRE_THROWS_AS(g.unlock(), std::runtime_error);
}
TEST_CASE("passes_while_unlocked") {
    turnstile::gate g {};
    g.unlock();
    REQUIRE_THROWS_AS(g.passes(), std::runtime_error);
}
TEST_CASE("pass_locks_again") {
    turnstile::gate g {};
    g.unlock();
    g.pass();
    REQUIRE_THROWS_AS(g.pass(), std::runtime_error);
    REQUIRE(g.passes() == 1);
}
TEST_CASE("a_failed_pass_changes_nothing") {
    turnstile::gate g {};
    REQUIRE_THROWS_AS(g.pass(), std::runtime_error);
    REQUIRE(g.passes() == 0);
}
TEST_CASE("many_passes") {
    turnstile::gate g {};
    for (int i {0}; i < 1000; ++i) {
        g.unlock();
        g.pass();
    }
    REQUIRE(g.passes() == 1000);
}
```
