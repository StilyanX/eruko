# 52.02
Source: written
Write a class `time_of_day::clock_time` whose constructor takes an hour and a minute, each an `int`.
The constructor throws `std::domain_error` when the hour is not from 0 to 23 or the minute is not from 0 to 59.
`hour()` and `minute()` return them.
`minutes_since_midnight()` returns how many minutes have passed since 0:00.
Limits: -1000 ≤ hour, minute ≤ 1000.
## Solution
```cpp
#pragma once

#include <stdexcept>

namespace time_of_day {

//: Write a class that keeps its data in private members.
class clock_time {
    int hour_ [[{0}]];
    int minute_ {0};
public:
    //: Write a constructor that sets the data, and stops with an error when it is bad.
    clock_time(int hour, int minute)
        : hour_ {[[hour]]}, minute_ {minute}
    {
        //:: Stop with an error when the hour is not from 0 to 23 or the minute is not from 0 to 59.
        if (hour < 0 || hour > 23 || minute < 0
            || [[minute > 59]]) {
            throw std::domain_error {"not a time of day"};
        }
    }
    //: Write member functions that give back the data.
    int hour() const
    {
        return [[hour_]];
    }
    int minute() const
    {
        return minute_;
    }
    //: Write a member function that gives back a value worked out in one step.
    int minutes_since_midnight() const
    {
        return [[hour_ * 60 + minute_]];
    }
};

}  // namespace time_of_day
```
## Slow
```cpp
#pragma once

#include <stdexcept>

namespace time_of_day {

class clock_time {
    int hour_ {0};
    int minute_ {0};
public:
    clock_time(int hour, int minute)
    {
        bool found {false};
        for (int h {0}; h < 24; ++h) {
            for (int m {0}; m < 60; ++m) {
                if (h == hour && m == minute) {
                    found = true;
                }
            }
        }
        if (!found) {
            throw std::domain_error {"bad"};
        }
        hour_ = hour;
        minute_ = minute;
    }
    int hour() const
    {
        return hour_;
    }
    int minute() const
    {
        return minute_;
    }
    int minutes_since_midnight() const
    {
        int count {0};
        for (int h {0}; h < hour_; ++h) {
            count += 60;
        }
        for (int m {0}; m < minute_; ++m) {
            ++count;
        }
        return count;
    }
};

}  // namespace time_of_day
```
## Tests
```cpp
TEST_CASE("morning") {
    const time_of_day::clock_time time {7, 30};
    REQUIRE(time.minutes_since_midnight() == 450);
}
TEST_CASE("hour and minute") {
    const time_of_day::clock_time time {13, 5};
    REQUIRE(time.hour() == 13);
    REQUIRE(time.minute() == 5);
}
TEST_CASE("midnight") {
    const time_of_day::clock_time time {0, 0};
    REQUIRE(time.minutes_since_midnight() == 0);
}
TEST_CASE("last minute") {
    const time_of_day::clock_time time {23, 59};
    REQUIRE(time.minutes_since_midnight() == 1439);
}
TEST_CASE("noon") {
    const time_of_day::clock_time time {12, 0};
    REQUIRE(time.minutes_since_midnight() == 720);
}
TEST_CASE("hour 24") { REQUIRE_THROWS_AS((time_of_day::clock_time {24, 0}), std::domain_error); }
```
