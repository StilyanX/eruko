# 35.02
Source: written
Write a function `bell::rows(std::size_t n)` that returns the first n rows of a triangle of numbers as a `std::vector<std::vector<int>>`.
Row 1 is `{1}`. Row k holds k values: its first value is the last value of the row before, and each value after it is the value before it plus the value of the row before at the index before.
For n = 0 it returns an empty list.
Limits: 0 ≤ n ≤ 10.
## Solution
```cpp
#pragma once

#include <cstddef>
#include <vector>

namespace bell {

//: Write a function that builds each row from the row before it.
std::vector<std::vector<int>> rows([[std::size_t n]])
{
    //:: Start the result as an empty list.
    std::vector<std::vector<int>> result [[{}]];
    //:: Add each row, starting with the last value of the row before, each next value the one before it plus the one above that.
    for (std::size_t length {1}; length <= n; ++length) {
        std::vector<int> row {1};
        if (length > 1) {
            row = {result.back().back()};
        }
        for (std::size_t i {1}; i < length; ++i) {
            row.push_back([[row.back() + result.back()[i - 1]]]);
        }
        result.push_back(row);
    }
    //:: Give back the result.
    return [[result]];
}

}  // namespace bell
```
## Slow
```cpp
#pragma once

#include <cstddef>
#include <vector>

namespace bell {

inline int value(std::size_t row, std::size_t column)
{
    if (row == 0) {
        return 1;
    }
    if (column == 0) {
        return value(row - 1, row - 1);
    }
    return value(row, column - 1) + value(row - 1, column - 1);
}

inline std::vector<std::vector<int>> rows(std::size_t n)
{
    std::vector<std::vector<int>> result {};
    for (std::size_t row {0}; row < n; ++row) {
        std::vector<int> values {};
        for (std::size_t column {0}; column <= row; ++column) {
            values.push_back(value(row, column));
        }
        result.push_back(values);
    }
    return result;
}

}  // namespace bell
```

TEST_CASE("four_rows") {
    const std::vector<std::vector<int>> expected {
        {1}, {1, 2}, {2, 3, 5}, {5, 7, 10, 15}};
    REQUIRE(bell::rows(4) == expected);
}
TEST_CASE("no_rows") { REQUIRE(bell::rows(0).empty()); }
TEST_CASE("one_row") {
    const std::vector<std::vector<int>> expected {{1}};
    REQUIRE(bell::rows(1) == expected);
}
TEST_CASE("two_rows") {
    const std::vector<std::vector<int>> expected {{1}, {1, 2}};
    REQUIRE(bell::rows(2) == expected);
}
TEST_CASE("ten_rows_last_row") {
    const std::vector<int> expected {
        21147, 25287, 30304, 36401, 43833,
        52922, 64077, 77821, 94828, 115975};
    const std::vector<std::vector<int>> all {bell::rows(10)};
    REQUIRE(all.size() == 10);
    REQUIRE(all.back() == expected);
}
TEST_CASE("each_row_is_one_longer") {
    const std::vector<std::vector<int>> all {bell::rows(7)};
    for (std::size_t index {0}; index < all.size(); ++index) {
        REQUIRE(all[index].size() == index + 1);
    }
}
