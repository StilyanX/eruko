# 29.02
Source: written
Write a function `multiples::sum_of_threes(int n)` that returns the sum of the multiples of 3 from 1 to n.
Write a function `multiples::sum_of_fives(int n)` that returns the sum of the multiples of 5 from 1 to n.
Write a function `multiples::both(int n)` that returns the first of these plus the second, so a multiple of 15 counts twice.
Limits: 1 ≤ n ≤ 1000.
## Solution
```cpp
#pragma once

namespace multiples {

//: Write a function that adds up a sequence and gives back the total.
int sum_of_threes([[int n]])
{
    //:: Start the total at zero.
    int total {[[0]]};
    //:: Add each multiple of 3 from 3 to n to the total.
    for (int number {3}; number <= n; number += 3) {
        total += [[number]];
    }
    //:: Give back the total.
    return [[total]];
}

//: Write a function that adds up a sequence and gives back the total.
int sum_of_fives([[int n]])
{
    //:: Start the total at zero.
    int total {[[0]]};
    //:: Add each multiple of 5 from 5 to n to the total.
    for (int number {5}; number <= n; number += 5) {
        total += [[number]];
    }
    //:: Give back the total.
    return [[total]];
}

//: Write a function that calls the functions above and gives back the answer.
int both([[int n]])
{
    return sum_of_threes(n) + sum_of_fives(n);
}

}  // namespace multiples
```

#pragma once

namespace multiples {

inline int sum_of_threes(int n)
{
    int total {0};
    for (int number {1}; number <= n; ++number) {
        if (number % 3 == 0) {
            total += number;
        }
    }
    return total;
}

inline int sum_of_fives(int n)
{
    int total {0};
    for (int number {1}; number <= n; ++number) {
        if (number % 5 == 0) {
            total += number;
        }
    }
    return total;
}

inline int both(int n)
{
    int total {0};
    for (int number {1}; number <= n; ++number) {
        if (number % 3 == 0) {
            total += number;
        }
        if (number % 5 == 0) {
            total += number;
        }
    }
    return total;
}

}  // namespace multiples
