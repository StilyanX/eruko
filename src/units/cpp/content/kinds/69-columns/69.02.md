# 69.02
Source: written
Write a class `column_reader::reader` whose constructor takes a `const std::string& text` and a width as an `int`, and whose member `column(int index)` returns a `std::string`.
The object keeps only the letters of text, in order, made uppercase.
Written row by row in rows of width letters, the last row maybe shorter, `column(index)` returns the letters of column index from top to bottom, counting the first column as 0.
Limits: 1 ≤ width ≤ 20; 0 ≤ index < width; 0 ≤ length of text ≤ 200; each character is ASCII.
## Solution
```cpp
#pragma once

#include <cctype>
#include <string>

namespace column_reader {

//: Write a class that keeps its data in private members.
class reader {
    std::string text_ [[{}]];
    int width_ {1};
public:
    //: Write a constructor that sets the data.
    reader(const std::string& text, int width)
        : width_ {[[width]]}
    {
        //:: Keep each letter of the text, made uppercase, in order.
        for (const unsigned char sign : text) {
            if ([[std::isalpha(sign)]]) {
                text_ += static_cast<char>(std::toupper(sign));
            }
        }
    }
    //: Write a member function that builds a new string one piece at a time and gives it back.
    std::string column([[int index]]) const
    {
        //:: Start the result as an empty string.
        std::string result {[[""]]};
        //:: Add every letter from the index on, one row apart.
        for (int at {index}; at < std::ssize(text_);
             at += width_) {
            result += [[text_[at]]];
        }
        //:: Give back the result.
        return [[result]];
    }
};

}  // namespace column_reader
```
## Slow
```cpp
#pragma once

#include <cctype>
#include <string>
#include <vector>

namespace column_reader {

class reader {
    std::vector<std::string> rows_ {};
public:
    reader(const std::string& text, int width)
    {
        std::string row {};
        for (const char sign : text) {
            if (std::isalpha(static_cast<unsigned char>(sign))) {
                row += static_cast<char>(std::toupper(sign));
                if (static_cast<int>(row.size()) == width) {
                    rows_.push_back(row);
                    row = "";
                }
            }
        }
        if (!row.empty()) {
            rows_.push_back(row);
        }
    }
    std::string column(int index) const
    {
        std::string result {};
        for (const std::string& row : rows_) {
            if (index < static_cast<int>(row.size())) {
                result += row[index];
            }
        }
        return result;
    }
};

}  // namespace column_reader
```
## Tests
```cpp
TEST_CASE("second_column") {
    const column_reader::reader grid {"abcdefg", 3};
    REQUIRE(grid.column(1) == "BE");
}
TEST_CASE("empty_text") {
    const column_reader::reader grid {"", 4};
    REQUIRE(grid.column(0) == "");
}
TEST_CASE("first_column_of_a_short_last_row") {
    const column_reader::reader grid {"abcdefg", 3};
    REQUIRE(grid.column(0) == "ADG");
}
TEST_CASE("last_column") {
    const column_reader::reader grid {"abcdefg", 3};
    REQUIRE(grid.column(2) == "CF");
}
TEST_CASE("only_letters_are_kept") {
    const column_reader::reader grid {"a1 b-c, d!E?f", 2};
    REQUIRE(grid.column(0) == "ACE");
    REQUIRE(grid.column(1) == "BDF");
}
TEST_CASE("width_1") {
    const column_reader::reader grid {"Hi there", 1};
    REQUIRE(grid.column(0) == "HITHERE");
}
TEST_CASE("width_larger_than_the_text") {
    const column_reader::reader grid {"abc", 20};
    REQUIRE(grid.column(1) == "B");
    REQUIRE(grid.column(19) == "");
}
TEST_CASE("text_with_no_letters") {
    const column_reader::reader grid {"123 456", 2};
    REQUIRE(grid.column(1) == "");
}
TEST_CASE("longest_text") {
    std::string text {};
    for (int i {0}; i < 200; ++i) {
        text += static_cast<char>('a' + i % 26);
    }
    const column_reader::reader grid {text, 20};
    REQUIRE(grid.column(0) == "AUOICWQKEY");
}
```
