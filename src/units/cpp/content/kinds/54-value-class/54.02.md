# 54.02
Source: written
Write a class `week::day` whose objects each hold a day of the week, numbered 0 for Monday to 6 for Sunday.
Its constructor takes a day number as an `int`, and keeps it turned into the range 0 to 6: 7 is 0 and -1 is 6.
`day::from(int number)` returns the day with that number.
`after(int days)` returns the day that many days later; a negative number goes earlier.
`name()` returns the day's name as a `std::string`: `Monday`, `Tuesday`, `Wednesday`, `Thursday`, `Friday`, `Saturday` or `Sunday`.
`==` and `!=` compare two days by their numbers.
Limits: -1000 ≤ number ≤ 1000; -1000 ≤ days ≤ 1000.
## Solution
```cpp
#pragma once

#include <array>
#include <string>

namespace week {

//: Write a class with a constructor that sets the data.
class day {
public:
    explicit day([[int number]])
        : number_ {(number % 7 + 7) % 7}
    {
    }
    //: Write a member function that gives back a value worked out in one step.
    static day from([[int number]])
    {
        return day {number};
    }
    //: Write a member function that gives back a value worked out in one step.
    day after(int days) const
    {
        return day {[[number_ + days]]};
    }
    //: Write a member function that gives back a value worked out in one step.
    std::string name() const
    {
        const std::array<std::string, 7> names {
            "Monday", "Tuesday", "Wednesday", "Thursday",
            "Friday", "Saturday", "Sunday"};
        return [[names[number_]]];
    }
    //: Compare 2 objects by their values with `==` and `!=`.
    bool operator==(const day& other) const = [[default]];
//: Keep the data in private members.
private:
    int number_ {[[0]]};
};

}  // namespace week
```
## Slow
```cpp
#pragma once

#include <string>

namespace week {

class day {
public:
    explicit day(int number) : number_ {number}
    {
        while (number_ < 0) {
            number_ += 7;
        }
        while (number_ > 6) {
            number_ -= 7;
        }
    }
    static day from(int number)
    {
        return day {number};
    }
    day after(int days) const
    {
        day result {number_};
        for (int step {0}; step < days; ++step) {
            result = day {result.number_ + 1};
        }
        for (int step {0}; step > days; --step) {
            result = day {result.number_ - 1};
        }
        return result;
    }
    std::string name() const
    {
        if (number_ == 0) {
            return "Monday";
        }
        if (number_ == 1) {
            return "Tuesday";
        }
        if (number_ == 2) {
            return "Wednesday";
        }
        if (number_ == 3) {
            return "Thursday";
        }
        if (number_ == 4) {
            return "Friday";
        }
        if (number_ == 5) {
            return "Saturday";
        }
        return "Sunday";
    }
    bool operator==(const day& other) const
    {
        return number_ == other.number_;
    }
private:
    int number_ {0};
};

}  // namespace week
```
