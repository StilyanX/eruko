# 59.02
Source: written
Write a function `signs::mix(const std::vector<int>& list)` that returns which signs the list holds, as a value of `enum class signs::Mix { both, positive, negative, none }`, which the header defines.
Return `both` when the list holds a number above 0 and a number below 0, `positive` when it holds only numbers above 0 among those not 0, `negative` when it holds only numbers below 0 among those not 0, else `none`.
Limits: 0 ≤ length of the list ≤ 1000; -1000 ≤ each number ≤ 1000.
## Solution
```cpp
#pragma once

#include <vector>

namespace signs {

//: Name each possible value in an `enum class`.
enum class Mix { [[both]], positive, negative, none };

//: Write a function that gives back true at the first item that passes a test, else false.
bool has_sign([[const std::vector<int>& list]], int sign)
{
    //:: Give back true at the first item whose sign is the sign given.
    for (const int item : list) {
        if ([[item * sign > 0]]) {
            return true;
        }
    }
    //:: Give back false, since no item matched.
    return [[false]];
}

//: Write a function that tests each case in order and gives back the result of the first that holds.
Mix mix([[const std::vector<int>& list]])
{
    //:: Give back both when the list holds both signs, positive or negative when it holds one, else none.
    const bool positive {has_sign(list, 1)};
    const bool negative {has_sign(list, -1)};
    if ([[positive && negative]]) {
        return Mix::both;
    } else if (positive) {
        return Mix::positive;
    } else if (negative) {
        return Mix::negative;
    } else {
        return Mix::none;
    }
}

}  // namespace signs
```
## Slow
```cpp
#pragma once

#include <vector>

namespace signs {

enum class Mix { both, positive, negative, none };

inline Mix mix(const std::vector<int>& list)
{
    int above {0};
    int below {0};
    for (const int item : list) {
        if (item > 0) {
            ++above;
        }
        if (item < 0) {
            ++below;
        }
    }
    Mix result {Mix::none};
    if (above > 0) {
        result = Mix::positive;
    }
    if (below > 0) {
        result = above > 0 ? Mix::both : Mix::negative;
    }
    return result;
}

}  // namespace signs
```

TEST_CASE("one_of_each") { REQUIRE(signs::mix({1, -2}) == signs::Mix::both); }
TEST_CASE("empty_list") { REQUIRE(signs::mix({}) == signs::Mix::none); }
TEST_CASE("only_zeros") { REQUIRE(signs::mix({0, 0, 0}) == signs::Mix::none); }
TEST_CASE("positive_with_zeros") {
    REQUIRE(signs::mix({0, 5, 0}) == signs::Mix::positive);
}
TEST_CASE("negative_only") {
    REQUIRE(signs::mix({-1000, -1}) == signs::Mix::negative);
}
TEST_CASE("one_positive") { REQUIRE(signs::mix({1000}) == signs::Mix::positive); }
TEST_CASE("one_negative") { REQUIRE(signs::mix({-7}) == signs::Mix::negative); }
TEST_CASE("both_far_apart") {
    std::vector<int> list(1000, 3);
    list.back() = -3;
    REQUIRE(signs::mix(list) == signs::Mix::both);
}
