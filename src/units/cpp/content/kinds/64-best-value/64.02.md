# 64.02
Source: written
Write a function `exact::best_exact(int capacity, const std::vector<exact::Item>& items)` that returns the largest sum of values of items you can choose, each item at most once, whose weights add up to exactly `capacity`, or -1 when no choice does.
Write also the struct `exact::Item`, with two `int` members, `weight` then `value`.
Choosing no item makes weight 0.
Limits: 0 ≤ items ≤ 20; 0 ≤ capacity ≤ 1000; 1 ≤ each weight ≤ 1000; 1 ≤ each value ≤ 1000.
## Solution
```cpp
#pragma once

#include <algorithm>
#include <vector>

namespace exact {

//: Write a struct that holds several values together.
struct Item {
    int weight {[[0]]};
    int value {0};
};

//: Write a function that keeps the best value for each weight, one item at a time.
int best_exact(int capacity,
               [[const std::vector<Item>& items]])
{
    //:: Start the best value for weight 0 at 0, and for each other weight at -1, the mark for none.
    std::vector<int> best(capacity + 1, [[-1]]);
    best[0] = 0;
    //:: For each item, keep the larger of the best value without it and with it, heaviest weight first, when the weight without it can be made.
    for (const Item& item : items) {
        for (int w {capacity}; w >= item.weight; --w) {
            const int rest {best[w - item.weight]};
            if ([[rest >= 0]]) {
                best[w] = std::max(best[w], rest + item.value);
            }
        }
    }
    //:: Give back the best value for the most weight.
    return [[best.back()]];
}

}  // namespace exact
```
## Slow
```cpp
#pragma once

#include <vector>

namespace exact {

struct Item {
    int weight {0};
    int value {0};
};

int best_exact(int capacity, const std::vector<Item>& items)
{
    const int n {static_cast<int>(items.size())};
    int answer {-1};
    for (int mask {0}; mask < (1 << n); ++mask) {
        int weight {0};
        int value {0};
        for (int i {0}; i < n; ++i) {
            if ((mask >> i) % 2 == 1) {
                weight += items[i].weight;
                value += items[i].value;
            }
        }
        if (weight == capacity && value > answer) {
            answer = value;
        }
    }
    return answer;
}

}  // namespace exact
```
## Tests
```cpp
TEST_CASE("three items") { REQUIRE(exact::best_exact(5, {{2, 3}, {3, 4}, {4, 6}}) == 7); }
TEST_CASE("no items and no capacity") { REQUIRE(exact::best_exact(0, {}) == 0); }
TEST_CASE("no items") { REQUIRE(exact::best_exact(3, {}) == -1); }
TEST_CASE("no capacity") { REQUIRE(exact::best_exact(0, {{1, 5}}) == 0); }
TEST_CASE("item too heavy") { REQUIRE(exact::best_exact(1, {{2, 5}}) == -1); }
TEST_CASE("one item fills it") { REQUIRE(exact::best_exact(4, {{4, 6}}) == 6); }
TEST_CASE("many small beat one large") { REQUIRE(exact::best_exact(6, {{1, 1}, {2, 2}, {3, 3}, {6, 5}}) == 6); }
TEST_CASE("the best part does not fill it") { REQUIRE(exact::best_exact(10, {{5, 10}, {5, 10}, {4, 100}}) == 20); }
TEST_CASE("cannot be made") { REQUIRE(exact::best_exact(7, {{2, 1}, {4, 1}, {6, 1}}) == -1); }
TEST_CASE("each item once") { REQUIRE(exact::best_exact(4, {{2, 9}}) == -1); }
TEST_CASE("largest") {
    const std::vector<exact::Item> items(20, exact::Item {50, 1000});
    REQUIRE(exact::best_exact(1000, items) == 20000);
}
TEST_CASE("largest that cannot be made") {
    const std::vector<exact::Item> items(20, exact::Item {2, 7});
    REQUIRE(exact::best_exact(999, items) == -1);
}
```
