# 49.02
Source: written
Write a function `capital_first::rotate(const std::string& phrase)` that returns, as a `std::string`, phrase with each word changed as follows, the words kept in order with one space between 2 words.
A word that holds a capital letter has its letters before the first capital moved to its end.
A word with no capital letter stays as it is.
Limits: phrase holds 1 to 50 words of letters, split by single spaces; each word is 1 to 20 letters.
## Solution
```cpp
#pragma once

#include <cctype>
#include <cstddef>
#include <sstream>
#include <string>

namespace capital_first {

//: Write a function that finds the first item that passes a test and gives back its position.
std::size_t split_point([[const std::string& word]])
{
    //:: Give back the index of the first capital letter, or 0 when there is none.
    for (std::size_t index {0}; index < word.size(); ++index) {
        if ([[std::isupper(word[index])]]) {
            return index;
        }
    }
    return 0;
}

//: Write a function that builds a new string one piece at a time and gives it back.
std::string rotate([[const std::string& phrase]])
{
    //:: Start the result as an empty string.
    std::string result {[[""]]};
    //:: For each word, add a space before all but the first, then the word from its split point and the letters before it.
    std::istringstream words {phrase};
    std::string word {};
    while (words >> word) {
        if (!result.empty()) {
            result += ' ';
        }
        const std::size_t split {split_point(word)};
        result += word.substr(split);
        result += [[word.substr(0, split)]];
    }
    //:: Give back the result.
    return [[result]];
}

}  // namespace capital_first
```
## Slow
```cpp
#pragma once

#include <string>
#include <vector>

namespace capital_first {

std::string rotate(const std::string& phrase)
{
    std::vector<std::string> words {""};
    for (const char character : phrase) {
        if (character == ' ') {
            words.push_back("");
        } else {
            words.back() += character;
        }
    }
    std::string result {""};
    for (const std::string& word : words) {
        std::string turned {word};
        for (std::size_t step {0}; step < word.size(); ++step) {
            if (turned[0] >= 'A' && turned[0] <= 'Z') {
                break;
            }
            turned = turned.substr(1) + turned[0];
        }
        if (turned[0] < 'A' || turned[0] > 'Z') {
            turned = word;
        }
        if (!result.empty()) {
            result += ' ';
        }
        result += turned;
    }
    return result;
}

}  // namespace capital_first
```

## Tests
```cpp
TEST_CASE("three_words")
{
    REQUIRE(capital_first::rotate("lowerUpper abC xyz")
            == "Upperlower Cab xyz");
}
TEST_CASE("one_small_letter") { REQUIRE(capital_first::rotate("a") == "a"); }
TEST_CASE("one_capital_letter") { REQUIRE(capital_first::rotate("A") == "A"); }
TEST_CASE("capital_at_the_start")
{
    REQUIRE(capital_first::rotate("Hello") == "Hello");
}
TEST_CASE("capital_at_the_end")
{
    REQUIRE(capital_first::rotate("abcD") == "Dabc");
}
TEST_CASE("two_capitals_split_at_the_first")
{
    REQUIRE(capital_first::rotate("abCdEf") == "CdEfab");
}
TEST_CASE("no_capital_in_any_word")
{
    REQUIRE(capital_first::rotate("some quiet words")
            == "some quiet words");
}
TEST_CASE("longest_phrase")
{
    std::string phrase {"aaaaaaaaaaaaaaaaaaaB"};
    std::string expected {"Baaaaaaaaaaaaaaaaaaa"};
    for (int index {1}; index < 50; ++index) {
        phrase += " aaaaaaaaaaaaaaaaaaaB";
        expected += " Baaaaaaaaaaaaaaaaaaa";
    }
    REQUIRE(capital_first::rotate(phrase) == expected);
}
```
