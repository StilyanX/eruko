# 62.02
Source: written
Write a class `jobs::job_queue` whose objects hold at most as many job names, each a `std::string`, as the `std::size_t` given to its constructor.
`next()` removes the name added first and returns it; when there is none it throws `std::domain_error`.
`submit(name)` adds the name last; when the queue is full it throws `std::domain_error`.
`submit_over(name)` adds the name last; when the queue is full it first removes the name added first.
`cancel_all()` removes every name.
Limits: 1 ≤ size given to the constructor ≤ 100.
## Solution
```cpp
#pragma once

#include <cstddef>
#include <deque>
#include <stdexcept>
#include <string>

namespace jobs {

//: Write a class that keeps its data in private members.
class job_queue {
    std::deque<std::string> names_ [[{}]];
    std::size_t capacity_ {0};
public:
    //: Write a constructor that sets the data.
    explicit job_queue(std::size_t capacity)
        : capacity_ {[[capacity]]}
    {
    }
    //: Write a member function that gives back the oldest item and removes it, or throws when there is none.
    std::string next()
    {
        if ([[names_.empty()]]) {
            throw std::domain_error {"no job is waiting"};
        }
        const std::string name {names_.front()};
        names_.pop_front();
        return name;
    }
    //: Write a member function that adds an item at the end, or throws when the buffer is full.
    void submit(const std::string& name)
    {
        if ([[names_.size() == capacity_]]) {
            throw std::domain_error {"the queue is full"};
        }
        names_.push_back(name);
    }
    //: Write a member function that adds an item at the end, removing the oldest first when the buffer is full.
    void submit_over(const std::string& name)
    {
        if (names_.size() == capacity_) {
            names_.[[pop_front()]];
        }
        names_.push_back(name);
    }
    //: Write a member function that removes every item.
    void cancel_all()
    {
        names_.[[clear()]];
    }
};

}  // namespace jobs
```
## Slow
```cpp
#pragma once

#include <cstddef>
#include <stdexcept>
#include <string>
#include <vector>

namespace jobs {

class job_queue {
public:
    explicit job_queue(std::size_t capacity)
        : capacity_ {capacity}
    {
    }
    std::string next()
    {
        if (names_.size() == 0) {
            throw std::domain_error {"empty"};
        }
        const std::string first {names_[0]};
        std::vector<std::string> rest {};
        for (std::size_t i {1}; i < names_.size(); ++i) {
            rest.push_back(names_[i]);
        }
        names_ = rest;
        return first;
    }
    void submit(const std::string& name)
    {
        if (names_.size() >= capacity_) {
            throw std::domain_error {"full"};
        }
        names_.push_back(name);
    }
    void submit_over(const std::string& name)
    {
        if (names_.size() >= capacity_) {
            next();
        }
        submit(name);
    }
    void cancel_all()
    {
        while (names_.size() > 0) {
            next();
        }
    }
private:
    std::vector<std::string> names_ {};
    std::size_t capacity_ {0};
};

}  // namespace jobs
```
## Tests
```cpp
TEST_CASE("submit_then_next") {
    jobs::job_queue queue {2};
    queue.submit("print");
    REQUIRE(queue.next() == "print");
}
TEST_CASE("next_on_empty_throws") {
    jobs::job_queue queue {1};
    REQUIRE_THROWS_AS(queue.next(), std::domain_error);
}
TEST_CASE("oldest_first") {
    jobs::job_queue queue {3};
    queue.submit("a");
    queue.submit("b");
    queue.submit("c");
    REQUIRE(queue.next() == "a");
    REQUIRE(queue.next() == "b");
    REQUIRE(queue.next() == "c");
    REQUIRE_THROWS_AS(queue.next(), std::domain_error);
}
TEST_CASE("submit_to_full_throws") {
    jobs::job_queue queue {1};
    queue.submit("a");
    REQUIRE_THROWS_AS(queue.submit("b"), std::domain_error);
    REQUIRE(queue.next() == "a");
}
TEST_CASE("room_again_after_next") {
    jobs::job_queue queue {1};
    queue.submit("a");
    REQUIRE(queue.next() == "a");
    queue.submit("b");
    REQUIRE(queue.next() == "b");
}
TEST_CASE("submit_over_when_room") {
    jobs::job_queue queue {2};
    queue.submit("a");
    queue.submit_over("b");
    REQUIRE(queue.next() == "a");
    REQUIRE(queue.next() == "b");
}
TEST_CASE("submit_over_when_full") {
    jobs::job_queue queue {2};
    queue.submit("a");
    queue.submit("b");
    queue.submit_over("c");
    REQUIRE(queue.next() == "b");
    REQUIRE(queue.next() == "c");
    REQUIRE_THROWS_AS(queue.next(), std::domain_error);
}
TEST_CASE("cancel_all_empties") {
    jobs::job_queue queue {2};
    queue.submit("a");
    queue.submit("b");
    queue.cancel_all();
    REQUIRE_THROWS_AS(queue.next(), std::domain_error);
    queue.submit("c");
    queue.submit("d");
    REQUIRE(queue.next() == "c");
}
TEST_CASE("largest") {
    jobs::job_queue queue {100};
    for (int i {0}; i < 100; ++i) {
        queue.submit(std::to_string(i));
    }
    REQUIRE_THROWS_AS(queue.submit("x"), std::domain_error);
    queue.submit_over("x");
    REQUIRE(queue.next() == "1");
}
```
