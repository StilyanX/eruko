# 65.02
Source: written
Write a function `base36::encode(const std::string& text, int a, int b)` that returns the text encoded with the key a, b.
Write a function `base36::decode(const std::string& text, int a, int b)` that returns the text decoded with the same key.
Number the 36 symbols in this order: the letters a = 0 to z = 25, in either case, then the digits 0 = 26 to 9 = 35. Encoding turns symbol number i into the symbol number (a × i + b) mod 36, a letter in lower case.
Decoding turns symbol number y into the symbol number x × (y - b) mod 36, where x is the number from 1 to 35 for which a × x mod 36 is 1.
Both leave out every character that is not a letter or a digit. Encoding then puts a dash after every 4 characters, but not at the end.
Throw `std::invalid_argument` when 2 or 3 divides a.
Limits: 1 ≤ a ≤ 100; 0 ≤ b ≤ 100; 0 ≤ length of the text ≤ 1000; every character is ASCII.
## Solution
```cpp
#pragma once

#include <cctype>
#include <numeric>
#include <stdexcept>
#include <string>

namespace base36 {

//: Write a function that throws an error when its input is not allowed.
void check_key([[int a]])
{
    if (std::gcd(a, 36) != 1) {
        throw std::invalid_argument {"a has no inverse"};
    }
}

//: Write a function that repeats a step until a condition holds and gives back the result.
int inverse([[int a]])
{
    //:: Start the inverse at 1.
    int inverse {[[1]]};
    //:: Add one to the inverse until a times it leaves 1 after dividing by 36.
    while ([[a * inverse % 36 != 1]]) {
        ++inverse;
    }
    //:: Give back the inverse.
    return [[inverse]];
}

//: Write a function that builds a new string from each character of the text.
std::string apply_key([[const std::string& text]], int times,
    int plus)
{
    //:: Start the result as an empty string.
    std::string result {[[""]]};
    //:: For each letter or digit, add the symbol the key moves it to.
    for (const char character : text) {
        if ([[std::isalnum(character)]]) {
            int index {std::tolower(character) - 'a'};
            if (std::isdigit(character)) {
                index = 26 + character - '0';
            }
            const int moved {(times * index + plus) % 36};
            if (moved < 26) {
                result += static_cast<char>('a' + moved);
            } else {
                result += static_cast<char>('0' + moved - 26);
            }
        }
    }
    //:: Give back the result.
    return [[result]];
}

//: Write a function that builds a new string from each character of the text.
std::string in_groups([[const std::string& text]])
{
    //:: Start the result as an empty string.
    std::string result {[[""]]};
    //:: For each character, add a dash before it when 4, 8, 12 or more characters come before it, then the character.
    for (std::size_t index {0}; index < text.size(); ++index) {
        if ([[index > 0 && index % 4 == 0]]) {
            result += '-';
        }
        result += text[index];
    }
    //:: Give back the result.
    return [[result]];
}

//: Write a function that calls the functions above and gives back the answer.
std::string encode([[const std::string& text, int a, int b]])
{
    check_key(a);
    return in_groups(apply_key(text, a, b));
}

//: Write a function that calls the functions above and gives back the answer.
std::string decode([[const std::string& text, int a, int b]])
{
    check_key(a);
    const int x {inverse(a)};
    return apply_key(text, x, x * (36 - b % 36));
}

}  // namespace base36
```
## Slow
```cpp
#pragma once

#include <stdexcept>
#include <string>

namespace base36 {

inline const std::string symbols {
    "abcdefghijklmnopqrstuvwxyz0123456789"};

inline int number_of(char character)
{
    for (int i {0}; i < 36; ++i) {
        const char lower {symbols[i]};
        const char upper {static_cast<char>(lower - 'a' + 'A')};
        if (character == lower || (i < 26 && character == upper)) {
            return i;
        }
    }
    return -1;
}

inline void check(int a)
{
    for (int x {1}; x < 36; ++x) {
        if (a * x % 36 == 1) {
            return;
        }
    }
    throw std::invalid_argument {"a has no inverse"};
}

inline std::string encode(const std::string& text, int a, int b)
{
    check(a);
    std::string plain {};
    for (const char character : text) {
        const int i {number_of(character)};
        if (i >= 0) {
            plain += symbols[(a * i + b) % 36];
        }
    }
    std::string result {};
    for (std::size_t k {0}; k < plain.size(); ++k) {
        if (k % 4 == 0 && k > 0) {
            result += '-';
        }
        result += plain[k];
    }
    return result;
}

inline std::string decode(const std::string& text, int a, int b)
{
    check(a);
    std::string result {};
    for (const char character : text) {
        const int y {number_of(character)};
        if (y < 0) {
            continue;
        }
        for (int i {0}; i < 36; ++i) {
            if ((a * i + b) % 36 == y) {
                result += symbols[i];
            }
        }
    }
    return result;
}

}  // namespace base36
```
## Tests
```cpp
TEST_CASE("encode_test") { REQUIRE(base36::encode("test", 5, 7) == "41z4"); }
TEST_CASE("encode_mixed_text") {
    REQUIRE(base36::encode("Hello, World 42", 7, 11) == "ydqq-bvbw-q6f1");
}
TEST_CASE("encode_empty") { REQUIRE(base36::encode("", 1, 0).empty()); }
TEST_CASE("encode_large_key") { REQUIRE(base36::encode("a", 35, 100) == "2"); }
TEST_CASE("encode_long_text") {
    REQUIRE(base36::encode("the quick brown fox 2026", 13, 5)
            == "ayv7-nb51-skhd-48hq-jtjz");
}
TEST_CASE("encode_wraps_to_the_start") {
    REQUIRE(base36::encode("zz99", 1, 1) == "00aa");
}
TEST_CASE("decode_test") { REQUIRE(base36::decode("41z4", 5, 7) == "test"); }
TEST_CASE("decode_with_dashes") {
    REQUIRE(base36::decode("ydqq-bvbw-q6f1", 7, 11) == "helloworld42");
}
TEST_CASE("decode_long_text") {
    REQUIRE(base36::decode("ayv7-nb51-skhd-48hq-jtjz", 13, 5)
            == "thequickbrownfox2026");
}
TEST_CASE("decode_empty") { REQUIRE(base36::decode("", 5, 0).empty()); }
TEST_CASE("encode_even_a_throws") {
    REQUIRE_THROWS_AS(base36::encode("test", 2, 1), std::invalid_argument);
}
TEST_CASE("decode_three_divides_a_throws") {
    REQUIRE_THROWS_AS(base36::decode("test", 9, 1), std::invalid_argument);
}
```
