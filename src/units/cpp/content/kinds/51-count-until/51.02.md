# 51.02
Source: written
Write a function `square_free::nth(int n)` that returns the nth whole number above 0 that no square of a whole number above 1 divides, counting 1 as the first.
Limits: 1 ≤ n ≤ 1000.
## Solution
```cpp
#pragma once

namespace square_free {

//: Write a function that gives back false at the first item that fails a test, else true.
bool is_square_free([[int number]])
{
    //:: Give back false at the first square from 4 up to the number that divides it.
    for (int root {2}; root * root <= number; ++root) {
        if ([[number % (root * root) == 0]]) {
            return false;
        }
    }
    //:: Give back true.
    return [[true]];
}

//: Write a function that repeats a step until a condition holds and gives back the result.
int nth([[int n]])
{
    //:: Start the number at 0 and the count at zero.
    int number {[[0]]};
    int count {0};
    //:: Add one to the number, and one to the count when no square divides it, until the count is n.
    while ([[count < n]]) {
        ++number;
        if (is_square_free(number)) {
            ++count;
        }
    }
    //:: Give back the number.
    return [[number]];
}

}  // namespace square_free
```
## Slow
```cpp
#pragma once

namespace square_free {

int nth(int n)
{
    int found {0};
    int candidate {0};
    while (found < n) {
        ++candidate;
        int squares {0};
        for (int divisor {2}; divisor <= candidate; ++divisor) {
            if (candidate % (divisor * divisor) == 0) {
                ++squares;
            }
        }
        if (squares == 0) {
            ++found;
        }
    }
    return candidate;
}

}  // namespace square_free
```
