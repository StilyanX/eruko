# 40.02
Source: written
Write a function `four_dice::score(std::array<int, 4> dice, const std::string& category)` that returns the score of the 4 dice in the category, as an `int`.
- `two pairs`: the sum of the dice when 2 show one face and the other 2 show another face.
- `three`: 3 times the face that at least 3 dice show.
- `all`: 40 when all 4 dice show the same face.
- `sum`: the sum of the dice.
Dice that do not fit the category score 0; 4 equal dice are not two pairs.
Limits: each die is from 1 to 6, in any order; the category is one of the 4 above.
## Solution
```cpp
#pragma once

#include <algorithm>
#include <array>
#include <string>

namespace four_dice {

//: Write a function that counts each item, then gives back the answer for the case it is given.
int score([[std::array<int, 4> dice]], const std::string& category)
{
    //:: Count the dice that show each face, and add each die to the total.
    std::array<int, 7> counts {};
    int total {0};
    for (const int die : dice) {
        ++counts[die];
        total += [[die]];
    }
    //:: Keep the face that the most dice show.
    int most {1};
    for (int face {2}; face <= 6; ++face) {
        if ([[counts[face] > counts[most]]]) {
            most = face;
        }
    }
    //:: Give back the score of each category when the dice fit it.
    const int top {counts[most]};
    if (category == "all" && [[top == 4]]) {
        return 40;
    } else if (category == "three" && top >= 3) {
        return 3 * most;
    } else if (category == "two pairs" && top == 2
               && std::ranges::count(counts, 2) == 2) {
        return total;
    } else if (category == "sum") {
        return total;
    }
    //:: Give back 0 when the dice do not fit the category.
    return [[0]];
}

}  // namespace four_dice
```
## Slow
```cpp
#pragma once

#include <algorithm>
#include <array>
#include <string>

namespace four_dice {

int score(std::array<int, 4> dice, const std::string& category)
{
    std::ranges::sort(dice);
    const int total {dice[0] + dice[1] + dice[2] + dice[3]};
    if (category == "sum") {
        return total;
    }
    if (category == "all" && dice[0] == dice[3]) {
        return 40;
    }
    if (category == "three" && dice[0] == dice[2]) {
        return 3 * dice[0];
    }
    if (category == "three" && dice[1] == dice[3]) {
        return 3 * dice[1];
    }
    if (category == "two pairs" && dice[0] == dice[1]
        && dice[2] == dice[3] && dice[1] != dice[2]) {
        return total;
    }
    return 0;
}

}  // namespace four_dice
```
