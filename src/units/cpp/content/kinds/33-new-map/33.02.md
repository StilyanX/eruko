# 33.02
Source: written
Write a function `invert::invert(const std::map<std::string, int>& codes)` that returns a `std::map<int, std::string>` from each value in codes to its key.
No 2 keys of codes have the same value.
Limits: 0 ≤ entries of codes ≤ 100; -1000 ≤ each value ≤ 1000; each key 1 to 10 lowercase letters.
## Solution
```cpp
#pragma once

#include <map>
#include <string>

namespace invert {

//: Write a function that builds a new map from each entry of the old one.
std::map<int, std::string> invert(
    [[const std::map<std::string, int>& codes]])
{
    //:: Start an empty map from each value to its key.
    std::map<int, std::string> result [[{}]];
    //:: Put each value in the result with its key.
    for (const auto& [key, value] : codes) {
        result[value] = [[key]];
    }
    //:: Give back the result.
    return [[result]];
}

}  // namespace invert
```
## Slow
```cpp
#pragma once

#include <map>
#include <string>

namespace invert {

std::map<int, std::string> invert(
    const std::map<std::string, int>& codes)
{
    std::map<int, std::string> result {};
    for (int value {-1000}; value <= 1000; ++value) {
        for (const auto& entry : codes) {
            if (entry.second == value) {
                result.insert({value, entry.first});
            }
        }
    }
    return result;
}

}  // namespace invert
```
## Tests
```cpp
TEST_CASE("two_entries") {
    const std::map<std::string, int> codes {{"red", 1}, {"blue", 2}};
    const std::map<int, std::string> expected {{1, "red"}, {2, "blue"}};
    REQUIRE(invert::invert(codes) == expected);
}
TEST_CASE("no_entries") {
    const std::map<std::string, int> codes {};
    REQUIRE(invert::invert(codes).empty());
}
TEST_CASE("one_entry") {
    const std::map<std::string, int> codes {{"a", 0}};
    const std::map<int, std::string> expected {{0, "a"}};
    REQUIRE(invert::invert(codes) == expected);
}
TEST_CASE("values_out_of_key_order") {
    const std::map<std::string, int> codes {
        {"apple", 30}, {"kiwi", -5}, {"pear", 7}, {"zucchini", 0}};
    const std::map<int, std::string> expected {
        {-5, "kiwi"}, {0, "zucchini"}, {7, "pear"}, {30, "apple"}};
    REQUIRE(invert::invert(codes) == expected);
}
TEST_CASE("smallest_and_largest_values") {
    const std::map<std::string, int> codes {
        {"low", -1000}, {"high", 1000}};
    const std::map<int, std::string> expected {
        {-1000, "low"}, {1000, "high"}};
    REQUIRE(invert::invert(codes) == expected);
}
TEST_CASE("hundred_entries") {
    std::map<std::string, int> codes {};
    std::map<int, std::string> expected {};
    for (int i {0}; i < 100; ++i) {
        const std::string key {static_cast<char>('a' + i / 26),
                               static_cast<char>('a' + i % 26)};
        codes[key] = 10 * i - 500;
        expected[10 * i - 500] = key;
    }
    REQUIRE(invert::invert(codes) == expected);
}
```
