# 60.01
Source: exercism list-ops
Write a function `list_ops::append(std::vector<T>& left, const std::vector<T>& right)` that adds the items of right, in order, to the end of left and returns nothing, and these function templates beside it in `list_ops`, each over a `std::vector<T>`, without the algorithms of the standard library:
`concat(const std::vector<std::vector<T>>& lists)` returns one list of the items of all the lists, in order.
`filter(const std::vector<T>& list, test)` returns the items for which `test(item)` is true, in order.
`length(const std::vector<T>& list)` returns how many items list holds, as a `std::size_t`.
`map(const std::vector<T>& list, function)` returns the list of `function(item)` for each item, in order.
`foldl(const std::vector<T>& list, T start, function)` starts a result at start and, for each item from first to last, makes the result `function(result, item)`; it returns the result.
`foldr` takes the same arguments and does the same, but takes the items from last to first.
`reverse(const std::vector<T>& list)` returns the items in the reverse order.
Limits: none
## Solution
```cpp
#pragma once

#include <cstddef>
#include <vector>

namespace list_ops {

//: Write a function that adds each item of one list to the end of another.
template <typename T>
void append(std::vector<T>& left, [[const std::vector<T>& right]])
{
    //:: Add each item of the right list, in order, to the end of the left.
    for (const T& item : right) {
        left.push_back([[item]]);
    }
}

//: Write a function that builds a new list one item at a time and gives it back.
template <typename T>
std::vector<T> concat([[const std::vector<std::vector<T>>& lists]])
{
    //:: Start the result as an empty list.
    std::vector<T> result [[{}]];
    //:: Add the items of each list, in order, to the end of the result.
    for (const std::vector<T>& list : lists) {
        append(result, [[list]]);
    }
    //:: Give back the result.
    return [[result]];
}

//: Write a function that builds a new list one item at a time and gives it back.
template <typename T, typename Test>
std::vector<T> filter([[const std::vector<T>& list]], Test test)
{
    //:: Start the result as an empty list.
    std::vector<T> result [[{}]];
    //:: Add each item, in order, when it passes the test.
    for (const T& item : list) {
        if ([[test(item)]]) {
            result.push_back(item);
        }
    }
    //:: Give back the result.
    return [[result]];
}

//: Write a function that adds one for each item and gives back the count.
template <typename T>
std::size_t length([[const std::vector<T>& list]])
{
    //:: Start the count at zero.
    std::size_t count {[[0]]};
    //:: Add one for each item.
    for (auto next {list.begin()}; next != list.end(); ++next) {
        ++[[count]];
    }
    //:: Give back the count.
    return [[count]];
}

//: Write a function that builds a new list one item at a time and gives it back.
template <typename T, typename Function>
std::vector<T> map([[const std::vector<T>& list]],
                   Function function)
{
    //:: Start the result as an empty list.
    std::vector<T> result [[{}]];
    //:: Add the function's answer for each item, in order.
    for (const T& item : list) {
        result.push_back([[function(item)]]);
    }
    //:: Give back the result.
    return [[result]];
}

//: Write a function that combines each item in turn into one result and gives it back.
template <typename T, typename Function>
T foldl(const std::vector<T>& list, [[T start]], Function function)
{
    //:: Start the result at the start value.
    T result {[[start]]};
    //:: Combine the result with each item, from first to last.
    for (const T& item : list) {
        result = [[function(result, item)]];
    }
    //:: Give back the result.
    return [[result]];
}

//: Write a function that combines each item in turn into one result and gives it back.
template <typename T, typename Function>
T foldr(const std::vector<T>& list, [[T start]], Function function)
{
    //:: Start the result at the start value.
    T result {[[start]]};
    //:: Combine the result with each item, from last to first.
    for (std::size_t index {list.size()}; index > 0; --index) {
        result = [[function(result, list[index - 1])]];
    }
    //:: Give back the result.
    return [[result]];
}

//: Write a function that builds a new list one item at a time and gives it back.
template <typename T>
std::vector<T> reverse([[const std::vector<T>& list]])
{
    //:: Start the result as an empty list.
    std::vector<T> result [[{}]];
    //:: Add each item, from last to first.
    for (std::size_t index {list.size()}; [[index > 0]]; --index) {
        result.push_back(list[index - 1]);
    }
    //:: Give back the result.
    return [[result]];
}

}  // namespace list_ops
```
