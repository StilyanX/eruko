# 60.02
Source: written
Write a function `vec_ops::extend(std::vector<T>& list, const std::vector<T>& more)` that adds the items of more, in order, to the end of list, and returns nothing.
Write these function templates in the same namespace too, each over a `std::vector<T>`, without the algorithms of the standard library:
`join(const std::vector<std::vector<T>>& lists, const T& between)` returns the items of all the lists, in order, with between placed once between each 2 lists next to each other.
`drop(const std::vector<T>& list, test)` returns the items for which `test(item)` is false, in order.
`size_of(const std::vector<T>& list)` returns how many items list holds, as a `std::size_t`.
`twice(const std::vector<T>& list, function)` returns the list of `function(function(item))` for each item, in order.
`fold_left(const std::vector<T>& list, T start, function)` starts a result at start and, for each item from first to last, makes the result `function(result, item)`; it returns the result.
`fold_right(const std::vector<T>& list, T start, function)` starts a result at start and, for each item from last to first, makes the result `function(item, result)`; it returns the result.
`every_second(const std::vector<T>& list)` returns the items at index 0, 2, 4 and so on, in order.
Limits: none
## Solution
```cpp
#pragma once

#include <cstddef>
#include <vector>

namespace vec_ops {

//: Write a function that adds each item of one list to the end of another.
template <typename T>
void extend(std::vector<T>& list, const std::vector<T>& more)
{
    for (const T& item : more) {
        list.push_back([[item]]);
    }
}

//: Write a function that builds a new list one item at a time and gives it back.
template <typename T>
std::vector<T> join(const std::vector<std::vector<T>>& lists,
                    const T& between)
{
    std::vector<T> result {};
    for (std::size_t index {0}; index < lists.size(); ++index) {
        if ([[index > 0]]) {
            result.push_back(between);
        }
        extend(result, lists[index]);
    }
    return result;
}

//: Write a function that builds a new list one item at a time and gives it back.
template <typename T, typename Test>
std::vector<T> drop(const std::vector<T>& list, Test test)
{
    std::vector<T> result {};
    for (const T& item : list) {
        if ([[!test(item)]]) {
            result.push_back(item);
        }
    }
    return result;
}

//: Write a function that adds one for each item and gives back the count.
template <typename T>
std::size_t size_of(const std::vector<T>& list)
{
    std::size_t count {0};
    for (auto next {list.begin()}; next != list.end(); ++next) {
        [[++count]];
    }
    return count;
}

//: Write a function that builds a new list one item at a time and gives it back.
template <typename T, typename Function>
std::vector<T> twice(const std::vector<T>& list,
                     Function function)
{
    std::vector<T> result {};
    for (const T& item : list) {
        result.push_back([[function(function(item))]]);
    }
    return result;
}

//: Write a function that combines each item in turn into one result and gives it back.
template <typename T, typename Function>
T fold_left(const std::vector<T>& list, T start,
            Function function)
{
    T result {start};
    for (const T& item : list) {
        result = [[function(result, item)]];
    }
    return result;
}

//: Write a function that combines each item in turn into one result and gives it back.
template <typename T, typename Function>
T fold_right(const std::vector<T>& list, T start,
             Function function)
{
    T result {start};
    for (std::size_t index {list.size()}; index > 0; --index) {
        result = [[function(list[index - 1], result)]];
    }
    return result;
}

//: Write a function that builds a new list one item at a time and gives it back.
template <typename T>
std::vector<T> every_second(const std::vector<T>& list)
{
    std::vector<T> result {};
    for (std::size_t at {0}; at < list.size(); [[at += 2]]) {
        result.push_back(list[at]);
    }
    return result;
}

}  // namespace vec_ops
```

## Slow
```cpp
#pragma once

#include <cstddef>
#include <vector>

namespace vec_ops {

template <typename T>
void extend(std::vector<T>& list, const std::vector<T>& more)
{
    const std::vector<T> copy {more};
    for (std::size_t index {0}; index < copy.size(); ++index) {
        list.insert(list.end(), copy[index]);
    }
}

template <typename T>
std::vector<T> join(const std::vector<std::vector<T>>& lists,
                    const T& between)
{
    std::vector<T> result {};
    bool first {true};
    for (const std::vector<T>& list : lists) {
        if (!first) {
            extend(result, std::vector<T> {between});
        }
        first = false;
        extend(result, list);
    }
    return result;
}

template <typename T, typename Test>
std::vector<T> drop(const std::vector<T>& list, Test test)
{
    std::vector<T> result {};
    for (std::size_t index {0}; index < list.size(); ++index) {
        if (test(list[index])) {
            continue;
        }
        result.push_back(list[index]);
    }
    return result;
}

template <typename T>
std::size_t size_of(const std::vector<T>& list)
{
    return list.size();
}

template <typename T, typename Function>
std::vector<T> twice(const std::vector<T>& list,
                     Function function)
{
    std::vector<T> once {};
    for (const T& item : list) {
        once.push_back(function(item));
    }
    std::vector<T> result {};
    for (const T& item : once) {
        result.push_back(function(item));
    }
    return result;
}

template <typename T, typename Function>
T fold_left(const std::vector<T>& list, T start,
            Function function)
{
    if (list.empty()) {
        return start;
    }
    const std::vector<T> rest {list.begin() + 1, list.end()};
    return fold_left(rest, function(start, list.front()), function);
}

template <typename T, typename Function>
T fold_right(const std::vector<T>& list, T start,
             Function function)
{
    if (list.empty()) {
        return start;
    }
    const std::vector<T> rest {list.begin(), list.end() - 1};
    return fold_right(rest, function(list.back(), start), function);
}

template <typename T>
std::vector<T> every_second(const std::vector<T>& list)
{
    std::vector<T> result {};
    for (std::size_t index {0}; index < list.size(); ++index) {
        if (index % 2 == 0) {
            result.push_back(list[index]);
        }
    }
    return result;
}

}  // namespace vec_ops
```
