# 42.02
Source: written
Write a function `bottles::recite(int start, int end)` that returns, as a `std::string`, the verses for the numbers from start down to end, with an empty line between 2 verses.
The verse for number n is one line: n in digits, then ` bottle` when n is 1 or ` bottles` otherwise, then ` left.` and a newline.
Limits: 1 ≤ end ≤ start ≤ 99.
## Solution
```cpp
#pragma once

#include <string>

namespace bottles {

//: Write a function that builds a new string one piece at a time and gives it back.
std::string verse([[int number]])
{
    //:: Start the result with the number in digits.
    std::string result {[[std::to_string(number)]]};
    //:: Add ` bottle` for 1, else ` bottles`.
    if ([[number == 1]]) {
        result += " bottle";
    } else {
        result += " bottles";
    }
    //:: Add ` left.` and a newline.
    result += [[" left.\n"]];
    //:: Give back the result.
    return [[result]];
}

//: Write a function that builds a new string one piece at a time and gives it back.
std::string recite([[int start]], int end)
{
    //:: Start the result as an empty string.
    std::string result {[[""]]};
    //:: For each number from start down to end, add an empty line before all but the first, then its verse.
    for (int number {start}; number >= end; --number) {
        if ([[number < start]]) {
            result += "\n";
        }
        result += verse(number);
    }
    //:: Give back the result.
    return [[result]];
}

}  // namespace bottles
```
## Slow
```cpp
#pragma once

#include <format>
#include <string>
#include <vector>

namespace bottles {

std::string recite(int start, int end)
{
    std::vector<std::string> verses {};
    for (int number {start}; number >= end; --number) {
        verses.push_back(std::format("{} bottle{} left.\n", number,
                                     number == 1 ? "" : "s"));
    }
    std::string result {};
    for (std::size_t index {0}; index < verses.size(); ++index) {
        if (index > 0) {
            result += "\n";
        }
        result += verses[index];
    }
    return result;
}

}  // namespace bottles
```
## Tests
```cpp
TEST_CASE("three_down_to_two") { REQUIRE(bottles::recite(3, 2) == "3 bottles left.\n\n2 bottles left.\n"); }
TEST_CASE("one_verse") { REQUIRE(bottles::recite(5, 5) == "5 bottles left.\n"); }
TEST_CASE("last_bottle") { REQUIRE(bottles::recite(1, 1) == "1 bottle left.\n"); }
TEST_CASE("two_down_to_one") { REQUIRE(bottles::recite(2, 1) == "2 bottles left.\n\n1 bottle left.\n"); }
TEST_CASE("largest_verse") { REQUIRE(bottles::recite(99, 99) == "99 bottles left.\n"); }
TEST_CASE("four_down_to_one") {
    REQUIRE(bottles::recite(4, 1) == "4 bottles left.\n\n3 bottles left.\n\n2 bottles left.\n\n1 bottle left.\n");
}
TEST_CASE("all_verses") {
    std::string expected {};
    for (int number {99}; number >= 1; --number) {
        expected += std::to_string(number) + (number == 1 ? " bottle" : " bottles") + " left.\n";
        if (number > 1) {
            expected += "\n";
        }
    }
    REQUIRE(bottles::recite(99, 1) == expected);
}
```
