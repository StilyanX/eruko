# 66.02
Source: written
Write a function `neighbours::spread(std::string digits)` that returns, as a `std::optional<std::string>`, the smallest order of the digits in which no 2 digits side by side are equal.
The answer keeps every digit, and may start with 0.
When no order works it returns `std::nullopt`.
Limits: 1 ≤ length of digits ≤ 8; each character of digits is a digit.
## Solution
```cpp
#pragma once

#include <algorithm>
#include <optional>
#include <string>

namespace neighbours {

//: Write a function that tries each order until one passes every test.
std::optional<std::string> spread([[std::string digits]])
{
    //:: Sort the digits, so the first order is the smallest.
    std::ranges::sort([[digits]]);
    //:: Give back the first order where no 2 digits side by side are equal.
    do {
        const auto equal {std::ranges::adjacent_find(digits)};
        if ([[equal == digits.end()]]) {
            return digits;
        }
    } while (std::ranges::next_permutation(digits).found);
    //:: Give back no answer when no order fits.
    return [[std::nullopt]];
}

}  // namespace neighbours
```
## Slow
```cpp
#pragma once

#include <optional>
#include <string>
#include <vector>

namespace neighbours {

inline void every_order(const std::string& digits, std::string& made,
                        std::vector<bool>& used,
                        std::optional<std::string>& best)
{
    if (made.size() == digits.size()) {
        bool fits {true};
        for (std::size_t i {1}; i < made.size(); ++i) {
            if (made[i] == made[i - 1]) {
                fits = false;
            }
        }
        if (fits && (!best || made < *best)) {
            best = made;
        }
        return;
    }
    for (std::size_t i {0}; i < digits.size(); ++i) {
        if (!used[i]) {
            used[i] = true;
            made.push_back(digits[i]);
            every_order(digits, made, used, best);
            made.pop_back();
            used[i] = false;
        }
    }
}

inline std::optional<std::string> spread(std::string digits)
{
    std::string made {};
    std::vector<bool> used(digits.size(), false);
    std::optional<std::string> best {};
    every_order(digits, made, used, best);
    return best;
}

}  // namespace neighbours
```
