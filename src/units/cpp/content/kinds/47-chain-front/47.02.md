# 47.02
Source: written
Write a class `plates::Stack` that holds plate weights as `int` values, where each element holds one weight and owns the next element, with these member functions.
`total() const` returns the sum of the weights held, as an `int`.
`put(int weight)` puts the weight on top.
`take()` takes the top weight off and returns it.
`top() const` returns the top weight and leaves it on.
`flip()` turns the order of the weights around, so the bottom weight is on top.
`take()` and `top()` throw `std::out_of_range` when the stack is empty.
A new stack is empty, with total 0.
Limits: 0 ≤ weights held ≤ 1000; 1 ≤ each weight ≤ 1000.
## Solution
```cpp
#pragma once

#include <memory>
#include <stdexcept>
#include <utility>

namespace plates {

//: Write a class that keeps its data in private members.
class Stack {
    struct Element {
        int weight {0};
        [[std::unique_ptr<Element>]] next {};
    };
    std::unique_ptr<Element> head {};
    int sum {0};
public:
    //: Write a member function that gives back the data.
    int total() const
    {
        return [[sum]];
    }
    //: Write a member function that puts a new element first.
    void put([[int weight]])
    {
        //:: Make the new element hold the value and the old first one.
        head = [[std::make_unique<Element>]](weight,
            std::move(head));
        //:: Add the weight to the total.
        sum += [[weight]];
    }
    //: Write a member function that takes off the first element and gives back its value.
    [[int]] take()
    {
        //:: Keep the value of the first element.
        const int weight {[[top()]]};
        //:: Make the next element the first, and take its weight from the total.
        head = [[std::move(head->next)]];
        sum -= weight;
        //:: Give back the value.
        return [[weight]];
    }
    //: Write a member function that gives back the value of the first element.
    [[int]] top() const
    {
        //:: Throw `std::out_of_range` when the list is empty.
        if ([[!head]]) {
            throw std::out_of_range {"the stack is empty"};
        }
        //:: Give back the value of the first element.
        return [[head->weight]];
    }
    //: Write a member function that turns the order of the elements around.
    [[void]] flip()
    {
        //:: Start the turned list empty.
        std::unique_ptr<Element> done [[{}]];
        //:: Move each first element to the front of the turned list.
        while ([[head]]) {
            auto rest {std::move(head->next)};
            head->next = std::move(done);
            done = std::move(head);
            head = std::move(rest);
        }
        //:: Make the turned list the list.
        head = [[std::move(done)]];
    }
};

}  // namespace plates
```
## Slow
```cpp
#pragma once

#include <stdexcept>
#include <vector>

namespace plates {

class Stack {
    std::vector<int> weights {};
public:
    int total() const
    {
        int sum {0};
        for (const int weight : weights) {
            sum += weight;
        }
        return sum;
    }
    void put(int weight)
    {
        weights.insert(weights.begin(), weight);
    }
    int take()
    {
        const int weight {top()};
        weights.erase(weights.begin());
        return weight;
    }
    int top() const
    {
        if (weights.empty()) {
            throw std::out_of_range {"the stack is empty"};
        }
        return weights.front();
    }
    void flip()
    {
        std::vector<int> turned {};
        for (const int weight : weights) {
            turned.insert(turned.begin(), weight);
        }
        weights = turned;
    }
};

}  // namespace plates
```
## Tests
```cpp
TEST_CASE("put_two_and_take_the_top") {
    plates::Stack stack {};
    stack.put(3);
    stack.put(5);
    REQUIRE(stack.total() == 8);
    REQUIRE(stack.take() == 5);
    REQUIRE(stack.total() == 3);
}
TEST_CASE("new_stack_is_empty") {
    const plates::Stack stack {};
    REQUIRE(stack.total() == 0);
}
TEST_CASE("take_from_empty_throws") {
    plates::Stack stack {};
    REQUIRE_THROWS_AS(stack.take(), std::out_of_range);
}
TEST_CASE("top_of_empty_throws") {
    const plates::Stack stack {};
    REQUIRE_THROWS_AS(stack.top(), std::out_of_range);
}
TEST_CASE("top_leaves_the_weight_on") {
    plates::Stack stack {};
    stack.put(7);
    REQUIRE(stack.top() == 7);
    REQUIRE(stack.top() == 7);
    REQUIRE(stack.total() == 7);
}
TEST_CASE("take_until_empty") {
    plates::Stack stack {};
    stack.put(1);
    stack.put(2);
    REQUIRE(stack.take() == 2);
    REQUIRE(stack.take() == 1);
    REQUIRE(stack.total() == 0);
    REQUIRE_THROWS_AS(stack.take(), std::out_of_range);
}
TEST_CASE("flip_puts_the_bottom_on_top") {
    plates::Stack stack {};
    stack.put(1);
    stack.put(2);
    stack.put(3);
    stack.flip();
    REQUIRE(stack.take() == 1);
    REQUIRE(stack.take() == 2);
    REQUIRE(stack.take() == 3);
}
TEST_CASE("flip_empty_and_one") {
    plates::Stack stack {};
    stack.flip();
    REQUIRE(stack.total() == 0);
    stack.put(4);
    stack.flip();
    REQUIRE(stack.top() == 4);
}
TEST_CASE("many_weights") {
    plates::Stack stack {};
    for (int weight {1}; weight <= 1000; ++weight) {
        stack.put(weight);
    }
    REQUIRE(stack.total() == 500500);
    stack.flip();
    REQUIRE(stack.top() == 1);
    REQUIRE(stack.take() == 1);
    REQUIRE(stack.total() == 500499);
}
```
