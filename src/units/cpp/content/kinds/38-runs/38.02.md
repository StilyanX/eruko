# 38.02
Source: written
Write a function `bracket_runs::encode(const std::string& text)` that returns the text with each run of the same character written as the character, then the run's length between `(` and `)`.
A run of length 1 is written as the character alone.
Write a function `bracket_runs::decode(const std::string& text)` that returns the text that `encode` turns into the given text.
Empty text gives empty text.
Limits: 0 ≤ length of text ≤ 1000; the text to encode holds letters and spaces only.
## Solution
```cpp
#pragma once

#include <cctype>
#include <cstddef>
#include <string>

namespace bracket_runs {

//: Write a function that builds a new string one piece at a time and gives it back.
std::string encode([[const std::string& text]])
{
    //:: Start the result as an empty string.
    std::string result {[[""]]};
    //:: For each run, add its character, then its length in brackets when above 1.
    std::size_t start {0};
    while (start < text.size()) {
        std::size_t end {start + 1};
        while (end < text.size() && text[end] == text[start]) {
            ++end;
        }
        result += text[start];
        if ([[end - start > 1]]) {
            result += "(" + std::to_string(end - start) + ")";
        }
        start = end;
    }
    //:: Give back the result.
    return [[result]];
}

//: Write a function that builds a new string from each character of the text.
std::string decode([[const std::string& text]])
{
    //:: Start the result as an empty string.
    std::string result {[[""]]};
    //:: Start the count at zero.
    int count {[[0]]};
    //:: For each character, add a digit to the count, add the last character count - 1 more times at `)`, else add any character but `(`.
    for (const char character : text) {
        if (std::isdigit(character)) {
            count = count * 10 + character - '0';
        } else if (character == ')') {
            result.append([[count - 1]], result.back());
            count = 0;
        } else if (character != '(') {
            result += character;
        }
    }
    //:: Give back the result.
    return [[result]];
}

}  // namespace bracket_runs
```
## Slow
```cpp
#pragma once

#include <string>

namespace bracket_runs {

std::string encode(const std::string& text)
{
    std::string result {};
    for (std::size_t i {0}; i < text.size(); ++i) {
        if (i > 0 && text[i] == text[i - 1]) {
            continue;
        }
        std::size_t length {0};
        for (std::size_t k {i}; k < text.size(); ++k) {
            if (text[k] != text[i]) {
                break;
            }
            ++length;
        }
        result += text[i];
        if (length > 1) {
            result += "(" + std::to_string(length) + ")";
        }
    }
    return result;
}

std::string decode(const std::string& text)
{
    std::string result {};
    std::size_t i {0};
    while (i < text.size()) {
        const char character {text[i]};
        ++i;
        int length {1};
        if (i < text.size() && text[i] == '(') {
            const std::size_t close {text.find(')', i)};
            length = std::stoi(text.substr(i + 1, close - i - 1));
            i = close + 1;
        }
        for (int k {0}; k < length; ++k) {
            result += character;
        }
    }
    return result;
}

}  // namespace bracket_runs
```

TEST_CASE("encode_runs") {
    REQUIRE(bracket_runs::encode("aaabccdddd") == "a(3)bc(2)d(4)");
}
TEST_CASE("encode_empty") { REQUIRE(bracket_runs::encode("") == ""); }
TEST_CASE("decode_empty") { REQUIRE(bracket_runs::decode("") == ""); }
TEST_CASE("encode_one_character") {
    REQUIRE(bracket_runs::encode("x") == "x");
}
TEST_CASE("encode_no_runs") {
    REQUIRE(bracket_runs::encode("abcd") == "abcd");
}
TEST_CASE("encode_one_run") {
    REQUIRE(bracket_runs::encode("zzzzz") == "z(5)");
}
TEST_CASE("encode_spaces_and_case") {
    REQUIRE(bracket_runs::encode("aA  Bbb") == "aA (2)Bb(2)");
}
TEST_CASE("encode_long_run") {
    REQUIRE(bracket_runs::encode(std::string(1000, 'q')) == "q(1000)");
}
TEST_CASE("decode_runs") {
    REQUIRE(bracket_runs::decode("a(3)bc(2)d(4)") == "aaabccdddd");
}
TEST_CASE("decode_no_runs") {
    REQUIRE(bracket_runs::decode("abcd") == "abcd");
}
TEST_CASE("decode_two_digit_run") {
    REQUIRE(bracket_runs::decode("k(12)m") == "kkkkkkkkkkkkm");
}
TEST_CASE("decode_spaces") {
    REQUIRE(bracket_runs::decode(" (3)x") == "   x");
}
TEST_CASE("decode_long_run") {
    REQUIRE(bracket_runs::decode("q(1000)") == std::string(1000, 'q'));
}
TEST_CASE("round_trip") {
    const std::string text {"  WWWWwww hhello   there zz"};
    REQUIRE(bracket_runs::decode(bracket_runs::encode(text)) == text);
}
