# 71.02
Source: written
Write a function `frog::fewest_jumps(int limit, int far, int near, int start, int goal)` that returns the fewest jumps that take a frog from cell start to cell goal on the cells 0 to limit.
Each jump goes forward far cells or back near cells, and must land on a cell from 0 to limit.
Write in namespace `frog` the type `enum class Outcome { reached, unreachable }` and the type `struct answer { Outcome outcome; int jumps; }`, which the function returns: `reached` and the fewest jumps, or `unreachable` and 0 when no jumps reach the goal.
Limits: 1 ≤ limit ≤ 100; 1 ≤ far ≤ limit; 1 ≤ near ≤ limit; 0 ≤ start ≤ limit; 0 ≤ goal ≤ limit.
## Solution
```cpp
#pragma once

#include <deque>
#include <set>
#include <vector>

namespace frog {

//: Name each possible value in an `enum class`.
enum class Outcome { [[reached]], unreachable };

//: Write a struct that holds several values together.
struct answer {
    Outcome outcome {[[Outcome::unreachable]]};
    int jumps {0};
};

//: Write a struct that holds several values together.
struct state {
    int cell {[[0]]};
    int jumps {0};
};

//: Write a function that gives back the next states from a state.
std::vector<state> next_states([[const state& now]], int far,
    int near)
{
    //:: Give back the states after a jump forward and a jump back, one jump later.
    return {{now.cell + far, [[now.jumps + 1]]},
            {now.cell - near, now.jumps + 1}};
}

//: Write a function that tries each state in order of moves, and gives back the first that reaches the goal.
answer fewest_jumps([[int limit]], int far, int near, int start,
    int goal)
{
    //:: Start the states to try and the values seen with the start.
    std::deque<state> to_try {{[[start]], 0}};
    std::set<int> seen {start};
    //:: While states are left, take the first, give back the answer when it holds the goal, else add each next state not yet seen.
    while ([[!to_try.empty()]]) {
        const state now {to_try.front()};
        to_try.pop_front();
        if (now.cell == goal) {
            return {Outcome::reached, now.jumps};
        }
        for (const state next : next_states(now, far, near)) {
            const int cell {next.cell};
            const bool fits {0 <= cell && cell <= limit};
            if (fits && !seen.contains(cell)) {
                seen.insert(cell);
                to_try.push_back(next);
            }
        }
    }
    //:: Give back unreachable when no state reaches the goal.
    return {[[Outcome::unreachable]], 0};
}

}  // namespace frog
```
## Slow
```cpp
#pragma once

#include <vector>

namespace frog {

enum class Outcome { reached, unreachable };

struct answer {
    Outcome outcome {Outcome::unreachable};
    int jumps {0};
};

inline answer fewest_jumps(int limit, int far, int near, int start,
    int goal)
{
    std::vector<int> best(limit + 1, -1);
    best[start] = 0;
    for (int round {0}; round <= limit; ++round) {
        for (int cell {0}; cell <= limit; ++cell) {
            if (best[cell] < 0) {
                continue;
            }
            for (const int to : {cell + far, cell - near}) {
                if (to < 0 || to > limit) {
                    continue;
                }
                if (best[to] < 0 || best[cell] + 1 < best[to]) {
                    best[to] = best[cell] + 1;
                }
            }
        }
    }
    if (best[goal] < 0) {
        return {Outcome::unreachable, 0};
    }
    return {Outcome::reached, best[goal]};
}

}  // namespace frog
```

TEST_CASE("forward_twice_back_once") {
    const frog::answer found {frog::fewest_jumps(10, 3, 2, 0, 1)};
    REQUIRE(found.outcome == frog::Outcome::reached);
    REQUIRE(found.jumps == 2);
}
