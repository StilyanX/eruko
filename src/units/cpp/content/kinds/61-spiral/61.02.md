# 61.02
Source: written
Write a function `rings::rings(int size)` that returns a square grid of `size` rows of `size` numbers each, as a `std::vector` of rows, each a `std::vector<int>`.
Each cell on the outer edge holds 1, each cell on the edge of the square just inside holds 2, and so on inward.
Size 0 gives the empty grid.
Limits: 0 ≤ size ≤ 50.
## Solution
```cpp
#pragma once

#include <algorithm>
#include <vector>

namespace rings {

//: Give a type a short name.
using matrix = [[std::vector<std::vector<int>>]];

//: Write a function that gives back a value worked out in one step.
int ring([[int size, int row, int column]])
{
    //:: Give back 1 more than the distance from the cell to the nearest edge.
    return std::min({row, column, size - 1 - row,
                     size - 1 - column}) + [[1]];
}

//: Write a function that fills a grid and gives it back.
matrix rings([[int size]])
{
    //:: Start the grid with size rows of size zeros.
    matrix grid(size, std::vector<int>([[size]], 0));
    //:: Fill each cell of each row with its value.
    for (int row {0}; row < size; ++row) {
        for (int column {0}; column < size; ++column) {
            grid[row][column] = [[ring(size, row, column)]];
        }
    }
    //:: Give back the grid.
    return [[grid]];
}

}  // namespace rings
```
## Slow
```cpp
#pragma once

#include <vector>

namespace rings {

std::vector<std::vector<int>> rings(int size)
{
    std::vector<std::vector<int>> grid {};
    for (int row {0}; row < size; ++row) {
        std::vector<int> line {};
        for (int column {0}; column < size; ++column) {
            int depth {0};
            while (row >= depth && row < size - depth
                   && column >= depth && column < size - depth) {
                ++depth;
            }
            line.push_back(depth);
        }
        grid.push_back(line);
    }
    return grid;
}

}  // namespace rings
```
