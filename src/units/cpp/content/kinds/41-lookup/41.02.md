# 41.02
Source: written
Write a function `weather::sky(std::string_view week, std::string_view day)` that returns the sky of a day as a value of `enum class weather::Sky { sunny, cloudy, rainy }`, which the header defines.
The days, in order, are mon, tue, wed, thu, fri, sat and sun; week holds one letter for each day in that order: `S` sunny, `C` cloudy, `R` rainy.
Limits: week holds 7 letters, each `S`, `C` or `R`; day is one of the 7 days.
## Solution
```cpp
#pragma once

#include <array>
#include <string_view>

namespace weather {

//: Name each possible value in an `enum class`.
enum class Sky { [[sunny]], cloudy, rainy };

//: Write a function that tests each case in order and gives back the result of the first that holds.
Sky sky_of([[char letter]])
{
    //:: Give back sunny for `S`, cloudy for `C`, else rainy.
    if ([[letter == 'S']]) {
        return Sky::sunny;
    } else if (letter == 'C') {
        return Sky::cloudy;
    } else {
        return Sky::rainy;
    }
}

//: Write a function that finds the first item that passes a test and gives back its position.
int position([[std::string_view day]])
{
    //:: Make the list of days, in order.
    const std::array<std::string_view, 7> days {
        [["mon"]], "tue", "wed", "thu", "fri", "sat", "sun"};
    //:: Give back the index of the first day in the list equal to the day, else -1.
    for (int index {0}; index < std::ssize(days); ++index) {
        if ([[days[index] == day]]) {
            return index;
        }
    }
    return -1;
}

//: Write a function that gives back a value worked out in one step.
Sky sky([[std::string_view week]], std::string_view day)
{
    //:: Give back the sky for the day's letter.
    return [[sky_of(week[position(day)])]];
}

}  // namespace weather
```
## Slow
```cpp
#pragma once

#include <string>
#include <string_view>

namespace weather {

enum class Sky { sunny, cloudy, rainy };

inline Sky sky(std::string_view week, std::string_view day)
{
    const std::string names {"montuewedthufrisatsun"};
    std::size_t at {0};
    while (names.substr(at, 3) != day) {
        at += 3;
    }
    const char letter {week[at / 3]};
    Sky result {Sky::rainy};
    if (letter == 'C') {
        result = Sky::cloudy;
    }
    if (letter == 'S') {
        result = Sky::sunny;
    }
    return result;
}

}  // namespace weather
```
