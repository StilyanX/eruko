# 63.02
Source: written
Write a class `playlist::Playlist` whose objects are lists of songs, each a `std::string`, in which each node links to the node before it and the node after it.
`add_last(song)` adds song at the end, and `add_first(song)` adds it at the start.
`take_last()` removes the last song and returns it; `take_first()` removes the first song and returns it.
`drop(song)` removes the first node that holds song and returns true, or returns false when no node holds it.
`length()` returns how many songs the list holds, as a `std::size_t`.
Limits: `take_last()` and `take_first()` are never called on an empty list; at most 1000 songs are added.
## Solution
```cpp
#pragma once

#include <cstddef>
#include <memory>
#include <string>

namespace playlist {

//: Write a class that keeps its data in private members.
class Playlist {
    struct Node {
        std::string song {};
        std::unique_ptr<Node> next {};
        Node* previous {nullptr};
    };
    std::unique_ptr<Node> head_ [[{}]];
    Node* tail_ {nullptr};
    std::size_t size_ {0};
    //: Write a member function that unlinks a node and gives back its value.
    std::string unlink([[Node* node]])
    {
        //:: Keep the node's value.
        const std::string song {[[node->song]]};
        //:: Point the next node back at the node before, or the tail when there is no next node.
        if ([[node->next != nullptr]]) {
            node->next->previous = node->previous;
        } else {
            tail_ = node->previous;
        }
        //:: Point the node before at the next node, or the head when there is no node before.
        if (node->previous != nullptr) {
            node->previous->next = [[std::move(node->next)]];
        } else {
            head_ = std::move(node->next);
        }
        //:: Take one from the size and give back the value.
        --size_;
        return [[song]];
    }
public:
    //: Write member functions that add a node at one end.
    void add_last([[const std::string& song]])
    {
        //:: Make the node after the tail, or the head when the list is empty.
        auto& place {tail_ != nullptr ? tail_->next : head_};
        place = std::make_unique<Node>(song, nullptr, [[tail_]]);
        tail_ = place.get();
        ++size_;
    }
    void add_first(const std::string& song)
    {
        //:: Make the node before the head, and the tail when the list is empty.
        head_ = std::make_unique<Node>(
            song, [[std::move(head_)]], nullptr);
        if (head_->next != nullptr) {
            head_->next->previous = head_.get();
        } else {
            tail_ = head_.get();
        }
        ++size_;
    }
    //: Write member functions that take a node off one end and give back its value.
    std::string take_last()
    {
        return unlink([[tail_]]);
    }
    std::string take_first()
    {
        return unlink(head_.get());
    }
    //: Write a member function that finds the first item that passes a test and gives back whether it found one.
    bool drop([[const std::string& song]])
    {
        //:: Start at the head and move on until the node holds the value or the list ends.
        Node* node {head_.get()};
        while (node != nullptr && [[node->song != song]]) {
            node = node->next.get();
        }
        //:: Give back false when no node holds it, else unlink the node and give back true.
        if ([[node == nullptr]]) {
            return false;
        }
        unlink(node);
        return true;
    }
    //: Write a member function that gives back the data.
    std::size_t length() const
    {
        return [[size_]];
    }
};

}  // namespace playlist
```
## Slow
```cpp
#pragma once

#include <cstddef>
#include <string>
#include <vector>

namespace playlist {

class Playlist {
    std::vector<std::string> songs_ {};
public:
    void add_last(const std::string& song)
    {
        songs_.push_back(song);
    }
    void add_first(const std::string& song)
    {
        std::vector<std::string> moved {song};
        for (const std::string& old : songs_) {
            moved.push_back(old);
        }
        songs_ = moved;
    }
    std::string take_last()
    {
        const std::string song {songs_.back()};
        songs_.pop_back();
        return song;
    }
    std::string take_first()
    {
        const std::string song {songs_.front()};
        std::vector<std::string> rest {};
        for (std::size_t i {1}; i < songs_.size(); ++i) {
            rest.push_back(songs_[i]);
        }
        songs_ = rest;
        return song;
    }
    bool drop(const std::string& song)
    {
        std::vector<std::string> rest {};
        bool found {false};
        for (const std::string& old : songs_) {
            if (!found && old == song) {
                found = true;
            } else {
                rest.push_back(old);
            }
        }
        songs_ = rest;
        return found;
    }
    std::size_t length() const
    {
        return songs_.size();
    }
};

}  // namespace playlist
```

## Tests
```cpp
TEST_CASE("add_two_then_take_the_first") {
    playlist::Playlist list {};
    list.add_last("intro");
    list.add_last("outro");
    REQUIRE(list.take_first() == "intro");
}
TEST_CASE("empty_list_has_length_0") {
    const playlist::Playlist list {};
    REQUIRE(list.length() == 0);
}
TEST_CASE("take_last_gives_the_last_added") {
    playlist::Playlist list {};
    list.add_last("a");
    list.add_last("b");
    REQUIRE(list.take_last() == "b");
    REQUIRE(list.take_last() == "a");
    REQUIRE(list.length() == 0);
}
TEST_CASE("add_first_puts_the_song_at_the_start") {
    playlist::Playlist list {};
    list.add_first("b");
    list.add_first("a");
    list.add_last("c");
    REQUIRE(list.take_first() == "a");
    REQUIRE(list.take_last() == "c");
    REQUIRE(list.take_first() == "b");
}
TEST_CASE("list_can_be_used_again_after_empty") {
    playlist::Playlist list {};
    list.add_first("x");
    REQUIRE(list.take_last() == "x");
    list.add_last("y");
    REQUIRE(list.take_first() == "y");
    REQUIRE(list.length() == 0);
}
TEST_CASE("drop_a_middle_song") {
    playlist::Playlist list {};
    list.add_last("a");
    list.add_last("b");
    list.add_last("c");
    REQUIRE(list.drop("b"));
    REQUIRE(list.length() == 2);
    REQUIRE(list.take_first() == "a");
    REQUIRE(list.take_first() == "c");
}
TEST_CASE("drop_only_the_first_copy") {
    playlist::Playlist list {};
    list.add_last("a");
    list.add_last("b");
    list.add_last("a");
    REQUIRE(list.drop("a"));
    REQUIRE(list.take_first() == "b");
    REQUIRE(list.take_first() == "a");
}
TEST_CASE("drop_the_first_and_the_last") {
    playlist::Playlist list {};
    list.add_last("a");
    list.add_last("b");
    list.add_last("c");
    REQUIRE(list.drop("a"));
    REQUIRE(list.drop("c"));
    REQUIRE(list.take_last() == "b");
}
TEST_CASE("drop_a_missing_song") {
    playlist::Playlist list {};
    REQUIRE_FALSE(list.drop("a"));
    list.add_last("b");
    REQUIRE_FALSE(list.drop("a"));
    REQUIRE(list.length() == 1);
}
TEST_CASE("thousand_songs") {
    playlist::Playlist list {};
    for (int i {0}; i < 1000; ++i) {
        list.add_last(std::to_string(i));
    }
    REQUIRE(list.length() == 1000);
    REQUIRE(list.take_first() == "0");
    REQUIRE(list.take_last() == "999");
    REQUIRE(list.drop("500"));
    REQUIRE(list.length() == 997);
}
```
