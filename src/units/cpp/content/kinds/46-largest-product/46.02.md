# 46.02
Source: written
Write a function `digit_product::largest_product(const std::vector<int>& numbers)` that returns the largest product of the digits of a number in the list.
The digits of 0 are the one digit 0.
Limits: 1 ≤ size of the list ≤ 100; 0 ≤ each number ≤ 999 999 999.
## Solution
```cpp
#pragma once

#include <algorithm>
#include <string>
#include <vector>

namespace digit_product {

//: Write a function that multiplies a sequence and gives back the product.
int product([[int number]])
{
    //:: Start the product at 1.
    int result {[[1]]};
    //:: Times the product by each digit of the number.
    for (const char digit : std::to_string(number)) {
        result *= [[digit - '0']];
    }
    //:: Give back the product.
    return [[result]];
}

//: Write a function that keeps the largest value and gives it back.
int largest_product([[const std::vector<int>& numbers]])
{
    //:: Start the largest at zero.
    int largest {[[0]]};
    //:: For each number, keep the product of its digits when it is larger.
    for (const int number : numbers) {
        largest = std::max(largest, [[product(number)]]);
    }
    //:: Give back the largest.
    return [[largest]];
}

}  // namespace digit_product
```
## Slow
```cpp
#pragma once

#include <algorithm>
#include <vector>

namespace digit_product {

int largest_product(const std::vector<int>& numbers)
{
    std::vector<int> products {};
    for (const int number : numbers) {
        int left {number};
        int result {left % 10};
        left /= 10;
        while (left > 0) {
            result *= left % 10;
            left /= 10;
        }
        products.push_back(result);
    }
    std::ranges::sort(products);
    return products.back();
}

}  // namespace digit_product
```
