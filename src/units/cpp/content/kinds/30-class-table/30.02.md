# 30.02
Source: written
Write a class `grades::score` whose constructor takes a number of points as an `int`.
Each of these grades has a least number of points: A 90, B 80, C 70, D 60.
`reaches(const std::string& grade) const` returns whether the points are at least the grade's least number.
`reached() const` returns a `std::vector<std::string>` of every grade the points reach, in alphabetical order.
Limits: 0 ≤ points ≤ 100; each grade given is one of the 4 above.
## Solution
```cpp
#pragma once

#include <map>
#include <string>
#include <vector>

namespace grades {

//: Make a table of fixed values.
const [[std::map<std::string, int>]] least {
    {"A", 90}, {"B", 80}, {"C", 70}, {"D", 60}};

//: Write a class that keeps its data in private members.
class score {
    int points {[[0]]};
public:
    //: Write a constructor that sets the data.
    explicit score([[int value]]) : points {value}
    {
    }
    //: Write a member function that gives back a value worked out in one step.
    bool reaches([[const std::string& grade]]) const
    {
        //:: Give back whether the points are at least the grade's least number.
        return [[points >= least.at(grade)]];
    }
    //: Write a member function that gives back the items that pass a test.
    [[std::vector<std::string>]] reached() const
    {
        //:: Start the result as an empty list.
        std::vector<std::string> result [[{}]];
        //:: Add each grade whose least number the points reach.
        for (const auto& [grade, number] : least) {
            if ([[points >= number]]) {
                result.push_back(grade);
            }
        }
        //:: Give back the result.
        return [[result]];
    }
};

}  // namespace grades
```
## Slow
```cpp
#pragma once

#include <string>
#include <utility>
#include <vector>

namespace grades {

class score {
public:
    explicit score(int value) : points_ {value}
    {
    }
    bool reaches(const std::string& grade) const
    {
        for (const auto& [name, number] : table()) {
            if (name == grade) {
                return points_ >= number;
            }
        }
        return false;
    }
    std::vector<std::string> reached() const
    {
        std::vector<std::string> result {};
        for (const auto& [name, number] : table()) {
            if (reaches(name)) {
                result.push_back(name);
            }
        }
        return result;
    }
private:
    static std::vector<std::pair<std::string, int>> table()
    {
        return {{"A", 90}, {"B", 80}, {"C", 70}, {"D", 60}};
    }
    int points_ {0};
};

}  // namespace grades
```

TEST_CASE("eighty_five_reaches_b") { REQUIRE(grades::score(85).reaches("B")); }
TEST_CASE("eighty_five_misses_a") { REQUIRE_FALSE(grades::score(85).reaches("A")); }
TEST_CASE("exactly_the_least_number") { REQUIRE(grades::score(70).reaches("C")); }
TEST_CASE("one_below_the_least_number") { REQUIRE_FALSE(grades::score(59).reaches("D")); }
TEST_CASE("zero_reaches_none") { REQUIRE(grades::score(0).reached().empty()); }
TEST_CASE("hundred_reaches_all") {
    const std::vector<std::string> expected {"A", "B", "C", "D"};
    REQUIRE(grades::score(100).reached() == expected);
}

    REQUIRE(grades::score(75).reached() == expected);
}

    const std::vector<std::string> expected {"D"};
