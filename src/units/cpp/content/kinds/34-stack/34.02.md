# 34.02
Source: written
Write a function `react::react(const std::string& text)` that returns the text left after removing, again and again, a letter and the same letter in the other case side by side, as `aA` or `Aa`.
2 equal characters side by side, as `aa`, stay.
Limits: 0 ≤ length of the text ≤ 1000; the text holds only letters.
## Solution
```cpp
#pragma once

#include <cctype>
#include <string>

namespace react {

//: Write a function that keeps a stack of characters, adding to and taking from its end, and gives back the answer.
std::string react([[const std::string& text]])
{
    //:: Start the kept characters as an empty string.
    std::string kept {[[""]]};
    //:: For each character, take off the last kept one when it is the same letter in the other case, else add it.
    for (const char letter : text) {
        if (!kept.empty() && kept.back() != letter
            && std::tolower(kept.back())
                   == std::tolower(letter)) {
            kept.[[pop_back()]];
        } else {
            kept += letter;
        }
    }
    //:: Give back the kept characters.
    return [[kept]];
}

}  // namespace react
```
## Slow
```cpp
#pragma once

#include <cctype>
#include <string>

namespace react {

std::string react(const std::string& text)
{
    std::string left {text};
    bool removed {true};
    while (removed) {
        removed = false;
        for (std::size_t i {0}; i + 1 < left.size(); ++i) {
            const char a {left[i]};
            const char b {left[i + 1]};
            if (a != b && std::tolower(a) == std::tolower(b)) {
                left.erase(i, 2);
                removed = true;
                break;
            }
        }
    }
    return left;
}

}  // namespace react
```
## Tests
```cpp
TEST_CASE("long chain") { REQUIRE(react::react("dabAcCaCBAcCcaDA") == "dabCBAcaDA"); }
TEST_CASE("empty") { REQUIRE(react::react("") == ""); }
TEST_CASE("one letter") { REQUIRE(react::react("a") == "a"); }
TEST_CASE("one pair") { REQUIRE(react::react("aA") == ""); }
TEST_CASE("capital first") { REQUIRE(react::react("Bb") == ""); }
TEST_CASE("equal letters stay") { REQUIRE(react::react("aa") == "aa"); }
TEST_CASE("nested pairs") { REQUIRE(react::react("abBA") == ""); }
TEST_CASE("no pair side by side") { REQUIRE(react::react("abAB") == "abAB"); }
TEST_CASE("same case stays") { REQUIRE(react::react("aabAAB") == "aabAAB"); }
TEST_CASE("pairs one after another") { REQUIRE(react::react("AaBbCc") == ""); }
TEST_CASE("left after pairs") { REQUIRE(react::react("xaAbBy") == "xy"); }
TEST_CASE("longest") { REQUIRE(react::react(std::string(1000, 'z')) == std::string(1000, 'z')); }
TEST_CASE("longest that vanishes") {
    std::string text {};
    for (int i {0}; i < 500; ++i) {
        text += "qQ";
    }
    REQUIRE(react::react(text) == "");
}
```
