# 43.02
Source: written
Write a function `first_at_least::find(const std::vector<int>& data, int value)` that returns the index, as an `int`, of the first number in the list that is at least the value.
The list is sorted from smallest to largest and may hold a number more than once.
When every number is below the value, or the list is empty, it returns the length of the list.
Limits: 0 ≤ length of the list ≤ 1000; -1000 ≤ each number, value ≤ 1000.
## Solution
```cpp
#pragma once

#include <vector>

namespace first_at_least {

//: Write a function that halves the range that can hold the value until it finds it.
int find(const std::vector<int>& data, [[int value]])
{
    //:: Start the range as the whole list.
    int low {0};
    int high {[[static_cast<int>(data.size())]]};
    //:: Keep the half that can hold the first number at least the value, until one index is left.
    while (low < high) {
        const int middle {low + (high - low) / 2};
        if ([[data[middle] < value]]) {
            low = middle + 1;
        } else {
            high = middle;
        }
    }
    //:: Give back the index left.
    return [[low]];
}

}  // namespace first_at_least
```
## Slow
```cpp
#pragma once

#include <vector>

namespace first_at_least {

int find(const std::vector<int>& data, int value)
{
    int index {0};
    while (index < static_cast<int>(data.size())
           && data[index] < value) {
        ++index;
    }
    return index;
}

}  // namespace first_at_least
```
## Tests
```cpp
TEST_CASE("value_inside_the_list")
{
    REQUIRE(first_at_least::find({1, 3, 5, 7}, 4) == 2);
}
TEST_CASE("empty_list") { REQUIRE(first_at_least::find({}, 5) == 0); }
TEST_CASE("one_number_below") { REQUIRE(first_at_least::find({2}, 3) == 1); }
TEST_CASE("one_number_equal") { REQUIRE(first_at_least::find({3}, 3) == 0); }
TEST_CASE("value_below_every_number")
{
    REQUIRE(first_at_least::find({4, 6, 8}, -1000) == 0);
}
TEST_CASE("value_above_every_number")
{
    REQUIRE(first_at_least::find({4, 6, 8}, 1000) == 3);
}
TEST_CASE("repeated_numbers_give_the_first")
{
    REQUIRE(first_at_least::find({1, 2, 2, 2, 2, 3}, 2) == 1);
}
TEST_CASE("value_equal_to_the_last")
{
    REQUIRE(first_at_least::find({-5, 0, 5, 10}, 10) == 3);
}
TEST_CASE("largest_list")
{
    std::vector<int> data {};
    for (int number {-1000}; number < 1000; number += 2) {
        data.push_back(number);
    }
    REQUIRE(first_at_least::find(data, 1000) == 1000);
    REQUIRE(first_at_least::find(data, 501) == 751);
    REQUIRE(first_at_least::find(data, -999) == 1);
}
```
