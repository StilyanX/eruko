# 70.02
Source: written
Write a function `drinks::solve()` that returns a `drinks::Answer`, a struct with 2 `std::string` members: `tea_color`, the color of the house that drinks tea, and `fish_color`, the color of the house with the fish.
There are 3 houses in a row, numbered 0 to 2 from the left. Each has one of 3 colors (`red`, `green`, `blue`), one of 3 pets (cat, dog, fish) and one of 3 drinks (tea, milk, water), and no two houses share any of these.
These 5 facts hold:
- The blue house is just right of the green house.
- The cat lives in the red house.
- The fish lives in the green house.
- The house with the dog drinks milk.
- The house that drinks water is next to the house with the cat.
Exactly one way of filling the houses fits every fact.
Limits: none.
## Solution
```cpp
#pragma once

#include <algorithm>
#include <array>
#include <cstdlib>
#include <string>

namespace drinks {

//: Write a struct that holds several values together.
struct Answer {
    [[std::string]] tea_color {};
    std::string fish_color {};
};

//: Name the place of each value in its list.
enum { red, green, blue };
enum { [[cat]], dog, fish };
enum { tea, milk, water };

//: Make a table of fixed values.
const std::array<std::string, 3> names {[["red"]], "green", "blue"};

//: Write a function that gives back a value worked out in one step.
std::string name_at(const std::array<int, 3>& color, [[int house]])
{
    //:: Give back the name of the color of the house.
    const auto place {[[std::ranges::find(color, house)]]};
    return names[place - color.begin()];
}

//: Write a function that gives back a value worked out in one step.
bool near([[int first, int second]])
{
    //:: Give back whether the 2 houses are side by side.
    return [[std::abs(first - second) == 1]];
}

//: Write a function that gives back a value worked out in one step.
bool next_order([[std::array<int, 3>& order]])
{
    //:: Change the list to its next order, and give back false after the last.
    return [[std::ranges::next_permutation(order)]].found;
}

//: Write a function that tries each order until one passes every test.
Answer solve()
{
    std::array<int, 3> color {[[0, 1, 2]]};
    std::array<int, 3> pet {0, 1, 2};
    std::array<int, 3> drink {0, 1, 2};
    //:: Try each order of the colors with blue just right of green.
    do {
        if ([[color[blue] != color[green] + 1]]) {
            continue;
        }
        //:: Try each order of the pets that fits the colors.
        do {
            if ([[pet[cat] != color[red]]]
                || pet[fish] != color[green]) {
                continue;
            }
            //:: Give back the answer for the first order of drinks that fits.
            do {
                if ([[drink[milk] == pet[dog]]]
                    && near(drink[water], pet[cat])) {
                    return Answer {name_at(color, drink[tea]),
                                   name_at(color, pet[fish])};
                }
            } while (next_order(drink));
        } while (next_order(pet));
    } while (next_order(color));
    return {};
}

}  // namespace drinks
```
## Slow
```cpp
#pragma once

#include <array>
#include <string>
#include <vector>

namespace drinks {

struct Answer {
    std::string tea_color {};
    std::string fish_color {};
};

Answer solve()
{
    const std::array<std::string, 3> names {"red", "green", "blue"};
    std::vector<std::array<int, 3>> orders {};
    for (int a {0}; a < 3; ++a) {
        for (int b {0}; b < 3; ++b) {
            for (int c {0}; c < 3; ++c) {
                if (a != b && b != c && a != c) {
                    orders.push_back({a, b, c});
                }
            }
        }
    }
    Answer answer {};
    for (const auto& house_color : orders) {
        for (const auto& house_pet : orders) {
            for (const auto& house_drink : orders) {
                bool fits {true};
                int cat_house {0};
                int dog_house {0};
                int milk_house {0};
                int water_house {0};
                for (int h {0}; h < 3; ++h) {
                    if (h + 1 < 3 && house_color[h] == 1
                        && house_color[h + 1] != 2) {
                        fits = false;
                    }
                    if (h == 2 && house_color[h] == 1) {
                        fits = false;
                    }
                    if (house_pet[h] == 0 && house_color[h] != 0) {
                        fits = false;
                    }
                    if (house_pet[h] == 2 && house_color[h] != 1) {
                        fits = false;
                    }
                    if (house_pet[h] == 0) {
                        cat_house = h;
                    }
                    if (house_pet[h] == 1) {
                        dog_house = h;
                    }
                    if (house_drink[h] == 1) {
                        milk_house = h;
                    }
                    if (house_drink[h] == 2) {
                        water_house = h;
                    }
                }
                const int apart {water_house - cat_house};
                if (dog_house != milk_house || (apart != 1 && apart != -1)) {
                    fits = false;
                }
                if (fits) {
                    for (int h {0}; h < 3; ++h) {
                        if (house_drink[h] == 0) {
                            answer.tea_color = names[house_color[h]];
                        }
                        if (house_pet[h] == 2) {
                            answer.fish_color = names[house_color[h]];
                        }
                    }
                }
            }
        }
    }
    return answer;
}

}  // namespace drinks
```
