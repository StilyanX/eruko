# 24.02
Source: written
Write a function `compare::order(int a, int b)` that returns how a stands to b.
The header also defines the result type, `enum class compare::ordering { less, equal, greater }`.
It returns `less` when a is below b, `equal` when a equals b, and `greater` when a is above b.
Limits: -1000 ≤ a ≤ 1000; -1000 ≤ b ≤ 1000.
## Solution
```cpp
#pragma once

namespace compare {

//: Name each possible value in an `enum class`.
enum class ordering { [[less]], equal, greater };

//: Write a function that tests each case in order and gives back the result of the first that holds.
ordering order([[int a, int b]])
{
    //:: Give back less when a is below b, equal when they are equal, else greater.
    if ([[a < b]]) {
        return ordering::less;
    } else if (a == b) {
        return ordering::equal;
    } else {
        return ordering::greater;
    }
}

}  // namespace compare
```
## Slow
```cpp
#pragma once

namespace compare {

enum class ordering { less, equal, greater };

ordering order(int a, int b)
{
    for (int step {0}; step <= 2000; ++step) {
        if (a + step == b && step > 0) {
            return ordering::less;
        }
        if (b + step == a && step > 0) {
            return ordering::greater;
        }
    }
    return ordering::equal;
}

}  // namespace compare
```

TEST_CASE("two_below_five") { REQUIRE(compare::order(2, 5) == compare::ordering::less); }
TEST_CASE("five_above_two") { REQUIRE(compare::order(5, 2) == compare::ordering::greater); }
TEST_CASE("equal_numbers") { REQUIRE(compare::order(7, 7) == compare::ordering::equal); }
TEST_CASE("both_zero") { REQUIRE(compare::order(0, 0) == compare::ordering::equal); }
TEST_CASE("negative_below_zero") { REQUIRE(compare::order(-1, 0) == compare::ordering::less); }
TEST_CASE("smallest_and_largest") { REQUIRE(compare::order(-1000, 1000) == compare::ordering::less); }
TEST_CASE("largest_and_smallest") { REQUIRE(compare::order(1000, -1000) == compare::ordering::greater); }
