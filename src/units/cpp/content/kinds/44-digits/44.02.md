# 44.02
Source: written
Write a function `binary_sum::add(const std::vector<int>& first, const std::vector<int>& second)` that returns the binary digits of the sum of the 2 numbers whose binary digits it is given.
The digits come first digit first, each 0 or 1; return the digits of the sum first digit first too, with no leading zeros.
Leading zeros in the input count for nothing, and an empty list makes the number 0. A sum of 0 gives the empty list.
Limits: 0 ≤ number of digits in each list ≤ 30.
## Solution
```cpp
#pragma once

#include <vector>

namespace binary_sum {

//: Write a function that adds up a sequence and gives back the total.
int to_number([[const std::vector<int>& digits]])
{
    //:: Start the total at zero.
    int total {[[0]]};
    //:: For each digit, times the total by 2 and add the digit.
    for (const int digit : digits) {
        total = [[total * 2 + digit]];
    }
    //:: Give back the total.
    return [[total]];
}

//: Write a function that repeats a step until a condition holds and gives back the result.
std::vector<int> to_digits([[int number]])
{
    //:: Start the result as an empty list.
    std::vector<int> result [[{}]];
    //:: While the number is above 0, put its last binary digit at the front of the result, then drop it.
    while ([[number > 0]]) {
        result.insert(result.begin(), number % 2);
        number /= 2;
    }
    //:: Give back the result.
    return [[result]];
}

//: Write a function that calls the functions above and gives back the answer.
std::vector<int> add([[const std::vector<int>& first]],
    const std::vector<int>& second)
{
    //:: Give back the digits of the sum of the 2 numbers.
    return to_digits([[to_number(first) + to_number(second)]]);
}

}  // namespace binary_sum
```
## Slow
```cpp
#pragma once

#include <algorithm>
#include <vector>

namespace binary_sum {

std::vector<int> add(const std::vector<int>& first,
    const std::vector<int>& second)
{
    std::vector<int> a(first.rbegin(), first.rend());
    std::vector<int> b(second.rbegin(), second.rend());
    std::vector<int> sum {};
    int carry {0};
    for (std::size_t i {0}; i < a.size() || i < b.size() || carry; ++i) {
        int column {carry};
        if (i < a.size()) {
            column += a[i];
        }
        if (i < b.size()) {
            column += b[i];
        }
        sum.push_back(column % 2);
        carry = column / 2;
    }
    while (!sum.empty() && sum.back() == 0) {
        sum.pop_back();
    }
    std::reverse(sum.begin(), sum.end());
    return sum;
}

}  // namespace binary_sum
```

## Tests
```cpp
TEST_CASE("three_plus_one") {
    REQUIRE(binary_sum::add({1, 1}, {1}) == std::vector<int> {1, 0, 0});
}
TEST_CASE("both_empty") {
    REQUIRE(binary_sum::add({}, {}).empty());
}
TEST_CASE("zeros_only") {
    REQUIRE(binary_sum::add({0, 0}, {0}).empty());
}
TEST_CASE("one_empty") {
    REQUIRE(binary_sum::add({}, {1, 0, 1}) == std::vector<int> {1, 0, 1});
}
TEST_CASE("leading_zeros") {
    REQUIRE(binary_sum::add({0, 0, 1}, {0, 1, 0})
            == std::vector<int> {1, 1});
}
TEST_CASE("one_plus_one") {
    REQUIRE(binary_sum::add({1}, {1}) == std::vector<int> {1, 0});
}
TEST_CASE("no_carry") {
    REQUIRE(binary_sum::add({1, 0, 1, 0}, {1, 0, 1})
            == std::vector<int> {1, 1, 1, 1});
}
TEST_CASE("largest") {
    const std::vector<int> ones(30, 1);
    std::vector<int> expected(31, 1);
    expected.back() = 0;
    REQUIRE(binary_sum::add(ones, ones) == expected);
}
```
