# 56.02
Source: written
Write a function `strength::rate(const std::string& text)` that returns one of three answers for the text.
When it holds at least 8 characters, a digit, a capital letter and a small letter: `strong`.
When it holds at least 2 of these 3: a digit, a capital letter, a small letter: `medium`.
Else: `weak`.
Limits: 0 ≤ length of text ≤ 1000; text holds printable ASCII characters.
## Solution
```cpp
#pragma once

#include <cctype>
#include <string>

namespace strength {

//: Write a function that keeps what it finds in each character of the text, then gives back the answer for it.
std::string rate([[const std::string& text]])
{
    //:: Start with no digit, capital or small letter found.
    bool digit {false};
    bool capital {false};
    bool small {[[false]]};
    //:: Keep whether a digit, a capital or a small letter comes.
    for (const char character : text) {
        if (std::isdigit(character)) {
            digit = true;
        } else if (std::isupper(character)) {
            capital = true;
        } else if ([[std::islower(character)]]) {
            small = true;
        }
    }
    //:: Test each case in order, and give back the answer of the first that holds.
    if ([[text.size() >= 8 && digit && capital && small]]) {
        return "strong";
    } else if (digit + capital + small >= 2) {
        return "medium";
    } else {
        return "weak";
    }
}

}  // namespace strength
```
## Slow
```cpp
#pragma once

#include <string>

namespace strength {

bool holds_one_of(const std::string& text, const std::string& set)
{
    for (const char c : text) {
        for (const char s : set) {
            if (c == s) {
                return true;
            }
        }
    }
    return false;
}

std::string rate(const std::string& text)
{
    const int kinds {
        holds_one_of(text, "0123456789")
        + holds_one_of(text, "ABCDEFGHIJKLMNOPQRSTUVWXYZ")
        + holds_one_of(text, "abcdefghijklmnopqrstuvwxyz")};
    if (kinds == 3 && text.size() >= 8) {
        return "strong";
    }
    if (kinds >= 2) {
        return "medium";
    }
    return "weak";
}

}  // namespace strength
```
