# 50.02
Source: written
Write a function `three_rails::encode(const std::string& text)` that returns the text's characters at the indexes that 4 divides, then those at the other even indexes, then those at the odd indexes, each part in order.
Write a function `three_rails::decode(const std::string& text)` that returns the text that `encode` turns into the given text.
Empty text gives empty text.
Limits: 0 ≤ length of text ≤ 1000.
## Solution
```cpp
#pragma once

#include <string>

namespace three_rails {

//: Write a function that tests each case in order and gives back the result of the first that holds.
int rail_of([[int index]])
{
    //:: Give back rail 2 for an odd index, rail 0 for one that 4 divides, else rail 1.
    if ([[index % 2 == 1]]) {
        return 2;
    }
    if (index % 4 == 0) {
        return 0;
    }
    return 1;
}

//: Write a function that builds a new string one piece at a time and gives it back.
std::string encode([[const std::string& text]])
{
    //:: Start the result as an empty string.
    std::string result {[[""]]};
    //:: For each rail, add each character on it, in order.
    for (int rail {0}; rail < 3; ++rail) {
        for (int index {0}; index < std::ssize(text); ++index) {
            if ([[rail_of(index) == rail]]) {
                result += text[index];
            }
        }
    }
    //:: Give back the result.
    return [[result]];
}

//: Write a function that puts each character of the text in its place on the rails, rail by rail.
std::string decode([[const std::string& text]])
{
    //:: Start the result as a copy of the text.
    std::string result {[[text]]};
    //:: Start the position in the text at zero.
    int next {[[0]]};
    //:: For each rail, put the next character of the text at each index on it.
    for (int rail {0}; rail < 3; ++rail) {
        for (int index {0}; index < std::ssize(text); ++index) {
            if (rail_of(index) == rail) {
                result[index] = [[text[next]]];
                ++next;
            }
        }
    }
    //:: Give back the result.
    return [[result]];
}

}  // namespace three_rails
```
## Slow
```cpp
#pragma once

#include <string>
#include <vector>

namespace three_rails {

std::vector<std::size_t> order(std::size_t size)
{
    std::vector<std::size_t> indexes {};
    for (std::size_t i {0}; i < size; i += 4) {
        indexes.push_back(i);
    }
    for (std::size_t i {2}; i < size; i += 4) {
        indexes.push_back(i);
    }
    for (std::size_t i {1}; i < size; i += 2) {
        indexes.push_back(i);
    }
    return indexes;
}

std::string encode(const std::string& text)
{
    std::string result {};
    for (const std::size_t i : order(text.size())) {
        result += text[i];
    }
    return result;
}

std::string decode(const std::string& text)
{
    const std::vector<std::size_t> indexes {order(text.size())};
    std::string result {};
    for (std::size_t place {0}; place < text.size(); ++place) {
        for (std::size_t k {0}; k < indexes.size(); ++k) {
            if (indexes[k] == place) {
                result += text[k];
            }
        }
    }
    return result;
}

}  // namespace three_rails
```
