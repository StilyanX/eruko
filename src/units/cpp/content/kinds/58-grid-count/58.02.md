# 58.02
Source: written
Write a function `side_count::annotate(const std::vector<std::string>& grid)` that returns a grid of the same size whose each square holds, as a digit, the number of `#` squares directly above, below, left and right of it.
The grid is a rectangle of rows of the same length; each square is `#` or `.`.
No rows give no rows, and empty rows stay empty.
Limits: 0 ≤ rows ≤ 100; 0 ≤ length of each row ≤ 100.
## Solution
```cpp
#pragma once

#include <cstdlib>
#include <string>
#include <vector>

namespace side_count {

//: Write a function that gives back a value worked out in one step.
bool is_wall(const std::vector<std::string>& grid,
             [[int r, int c]])
{
    //:: Give back whether the square is in the grid and holds a `#`.
    return r >= 0 && r < std::ssize(grid) && c >= 0
           && c < std::ssize(grid[r]) && [[grid[r][c] == '#']];
}

//: Write a function that counts the items that pass a test and gives back the count.
int walls_beside(const std::vector<std::string>& grid,
                 [[int row, int column]])
{
    //:: Start the count at zero.
    int count {[[0]]};
    //:: Add one for each wall directly above, below, left or right of it.
    for (int r {row - 1}; r <= row + 1; ++r) {
        for (int c {column - 1}; c <= column + 1; ++c) {
            const int apart {std::abs(r - row)
                             + std::abs(c - column)};
            if (apart == 1 && [[is_wall(grid, r, c)]]) {
                ++count;
            }
        }
    }
    //:: Give back the count.
    return [[count]];
}

//: Write a function that builds a new grid from the answer for each square of the old one.
std::vector<std::string> annotate(
    [[const std::vector<std::string>& grid]])
{
    //:: Start the result as a copy of the grid.
    std::vector<std::string> result {[[grid]]};
    //:: Put the count of walls beside each square in its place.
    for (int r {0}; r < std::ssize(grid); ++r) {
        for (int c {0}; c < std::ssize(grid[r]); ++c) {
            result[r][c] = '0' + [[walls_beside(grid, r, c)]];
        }
    }
    //:: Give back the result.
    return [[result]];
}

}  // namespace side_count
```
## Slow
```cpp
#pragma once

#include <string>
#include <vector>

namespace side_count {

std::vector<std::string> annotate(const std::vector<std::string>& grid)
{
    std::vector<std::string> result {grid};
    for (std::size_t r {0}; r < grid.size(); ++r) {
        for (std::size_t c {0}; c < grid[r].size(); ++c) {
            int count {0};
            for (std::size_t i {0}; i < grid.size(); ++i) {
                for (std::size_t j {0}; j < grid[i].size(); ++j) {
                    const bool same_row {i == r && (j + 1 == c || c + 1 == j)};
                    const bool same_column {j == c && (i + 1 == r || r + 1 == i)};
                    if ((same_row || same_column) && grid[i][j] == '#') {
                        ++count;
                    }
                }
            }
            result[r][c] = static_cast<char>('0' + count);
        }
    }
    return result;
}

}  // namespace side_count
```

```cpp
TEST_CASE("two by two") {
    const std::vector<std::string> grid {"#.", ".."};
    const std::vector<std::string> expected {"01", "10"};
    REQUIRE(side_count::annotate(grid) == expected);
}
TEST_CASE("no rows") {
    const std::vector<std::string> grid {};
    REQUIRE(side_count::annotate(grid).empty());
}
TEST_CASE("empty row") {
    const std::vector<std::string> grid {""};
    const std::vector<std::string> expected {""};
    REQUIRE(side_count::annotate(grid) == expected);
}
TEST_CASE("one wall") {
    const std::vector<std::string> grid {"#"};
    const std::vector<std::string> expected {"0"};
    REQUIRE(side_count::annotate(grid) == expected);
}
TEST_CASE("no walls") {
    const std::vector<std::string> grid {"...", "..."};
    const std::vector<std::string> expected {"000", "000"};
    REQUIRE(side_count::annotate(grid) == expected);
}
TEST_CASE("all walls") {
    const std::vector<std::string> grid {"###", "###", "###"};
    const std::vector<std::string> expected {"232", "343", "232"};
    REQUIRE(side_count::annotate(grid) == expected);
}
TEST_CASE("corners do not count") {
    const std::vector<std::string> grid {"#.#", "...", "#.#"};
    const std::vector<std::string> expected {"020", "202", "020"};
    REQUIRE(side_count::annotate(grid) == expected);
}
TEST_CASE("one row") {
    const std::vector<std::string> grid {"#.#"};
    const std::vector<std::string> expected {"020"};
    REQUIRE(side_count::annotate(grid) == expected);
}
TEST_CASE("one column") {
    const std::vector<std::string> grid {"#", ".", "#"};
    const std::vector<std::string> expected {"0", "2", "0"};
    REQUIRE(side_count::annotate(grid) == expected);
}
TEST_CASE("largest") {
    const std::vector<std::string> grid(100, std::string(100, '#'));
    const std::vector<std::string> result {side_count::annotate(grid)};
    REQUIRE(result[0][0] == '2');
    REQUIRE(result[0][50] == '3');
    REQUIRE(result[50][50] == '4');
    REQUIRE(result[99][99] == '2');
}
```
