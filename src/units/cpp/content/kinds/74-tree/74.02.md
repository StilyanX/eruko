# 74.02
Source: written
Write a class `word_tree::tree` whose objects are trees in which each node keeps a word, a `std::string`, and owns a left and a right branch.
Its explicit constructor takes a `const std::string&` and makes a tree of one node that holds it.
`insert(const std::string& word)` adds a node for the word: from the top node down, it goes to the left branch when the word is shorter than the node's word, else to the right, until it reaches an empty branch, where the new node goes.
`word()` returns the node's word; `before()` and `after()` return the left and the right branch as `const std::unique_ptr<tree>&`, empty when there is no branch.
`begin()` and `end()` let a range `for` take every word of the tree from shortest to longest, words of the same length in the order they were added.
Limits: 1 ≤ number of words ≤ 1000; 0 ≤ length of each word ≤ 20.
## Solution
```cpp
#pragma once

#include <memory>
#include <string>
#include <vector>

namespace word_tree {

//: Write a class that keeps its data in private members.
class tree {
    std::string word_ {};
    std::unique_ptr<tree> before_ [[{}]];
    std::unique_ptr<tree> after_ {};
    std::vector<std::string> sorted_ {};
public:
    //: Write a constructor that sets the data.
    explicit tree(const std::string& word)
        : word_ {[[word]]}
    {
    }
    //: Write member functions that give back the data.
    const std::string& word() const
    {
        return [[word_]];
    }
    const std::unique_ptr<tree>& before() const
    {
        return before_;
    }
    const std::unique_ptr<tree>& after() const
    {
        return after_;
    }
    //: Write a member function that solves its task by calling itself on a smaller part.
    void insert([[const std::string& word]])
    {
        //:: Pick the left branch for a shorter word, else the right.
        auto& branch {
            [[word.size() < word_.size()]] ? before_ : after_};
        //:: Put a new tree there when the branch is empty, else insert into the branch.
        if ([[branch == nullptr]]) {
            branch = std::make_unique<tree>(word);
        } else {
            branch->insert(word);
        }
    }
    //: Write a member function that solves its task by calling itself on a smaller part.
    void collect([[std::vector<std::string>& words]]) const
    {
        //:: Add the left branch's values, then the data, then the right branch's values.
        if (before_ != nullptr) {
            before_->collect(words);
        }
        words.push_back([[word_]]);
        if (after_ != nullptr) {
            after_->collect(words);
        }
    }
    //: Write member functions that give back the start and the end of the values in order.
    auto begin()
    {
        //:: Fill the list with the values in order, then give back its start.
        sorted_.clear();
        collect(sorted_);
        return [[sorted_.cbegin()]];
    }
    auto end() const
    {
        return [[sorted_.cend()]];
    }
};

}  // namespace word_tree
```
## Slow
```cpp
#pragma once

#include <memory>
#include <string>
#include <vector>

namespace word_tree {

class tree {
public:
    explicit tree(const std::string& word)
        : word_ {word}, added_ {word}
    {
    }
    const std::string& word() const
    {
        return word_;
    }
    const std::unique_ptr<tree>& before() const
    {
        return before_;
    }
    const std::unique_ptr<tree>& after() const
    {
        return after_;
    }
    void insert(const std::string& word)
    {
        added_.push_back(word);
        tree* node {this};
        while (true) {
            std::unique_ptr<tree>& next {
                word.size() < node->word_.size() ? node->before_
                                                 : node->after_};
            if (!next) {
                next.reset(new tree {word});
                return;
            }
            node = next.get();
        }
    }
    std::vector<std::string>::const_iterator begin()
    {
        sorted_.clear();
        std::vector<bool> taken(added_.size(), false);
        for (std::size_t round {0}; round < added_.size(); ++round) {
            std::size_t best {added_.size()};
            for (std::size_t i {0}; i < added_.size(); ++i) {
                if (!taken[i] && (best == added_.size()
                    || added_[i].size() < added_[best].size())) {
                    best = i;
                }
            }
            taken[best] = true;
            sorted_.push_back(added_[best]);
        }
        return sorted_.cbegin();
    }
    std::vector<std::string>::const_iterator end() const
    {
        return sorted_.cend();
    }
private:
    std::string word_ {};
    std::unique_ptr<tree> before_ {};
    std::unique_ptr<tree> after_ {};
    std::vector<std::string> added_ {};
    std::vector<std::string> sorted_ {};
};

}  // namespace word_tree
```

## Tests
```cpp
TEST_CASE("shortest_first") {
    word_tree::tree words {"pear"};
    words.insert("fig");
    words.insert("banana");
    std::vector<std::string> order {};
    for (const std::string& word : words) {
        order.push_back(word);
    }
    REQUIRE(order == std::vector<std::string> {"fig", "pear", "banana"});
}
TEST_CASE("one_word") {
    word_tree::tree words {"kiwi"};
    REQUIRE(words.word() == "kiwi");
    REQUIRE(words.before() == nullptr);
    REQUIRE(words.after() == nullptr);
    std::vector<std::string> order {};
    for (const std::string& word : words) {
        order.push_back(word);
    }
    REQUIRE(order == std::vector<std::string> {"kiwi"});
}
TEST_CASE("shorter_goes_left") {
    word_tree::tree words {"apple"};
    words.insert("ox");
    REQUIRE(words.before() != nullptr);
    REQUIRE(words.before()->word() == "ox");
    REQUIRE(words.after() == nullptr);
}
TEST_CASE("same_length_goes_right") {
    word_tree::tree words {"cat"};
    words.insert("dog");
    REQUIRE(words.before() == nullptr);
    REQUIRE(words.after()->word() == "dog");
}
TEST_CASE("longer_goes_right") {
    word_tree::tree words {"cat"};
    words.insert("horse");
    words.insert("zebra");
    REQUIRE(words.after()->word() == "horse");
    REQUIRE(words.after()->after()->word() == "zebra");
}
TEST_CASE("same_length_in_order_added") {
    word_tree::tree words {"bb"};
    words.insert("aa");
    words.insert("c");
    words.insert("dd");
    words.insert("e");
    std::vector<std::string> order {};
    for (const std::string& word : words) {
        order.push_back(word);
    }
    REQUIRE(order
            == std::vector<std::string> {"c", "e", "bb", "aa", "dd"});
}
TEST_CASE("empty_word") {
    word_tree::tree words {"a"};
    words.insert("");
    std::vector<std::string> order {};
    for (const std::string& word : words) {
        order.push_back(word);
    }
    REQUIRE(order == std::vector<std::string> {"", "a"});
}
TEST_CASE("largest") {
    word_tree::tree words {std::string(10, 'm')};
    for (int i {1}; i < 1000; ++i) {
        words.insert(std::string(static_cast<std::size_t>(i % 21), 'x'));
    }
    std::vector<std::string> order {};
    for (const std::string& word : words) {
        order.push_back(word);
    }
    REQUIRE(order.size() == 1000);
    REQUIRE(order.front().empty());
    REQUIRE(order.back() == std::string(20, 'x'));
    REQUIRE(order[479] == std::string(10, 'm'));
}
```
