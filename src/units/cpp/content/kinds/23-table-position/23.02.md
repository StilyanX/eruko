# 23.02
Source: written
Write a function `planets::names()` that returns the 8 planets in order from the sun, as a `std::vector<std::string>`: mercury, venus, earth, mars, jupiter, saturn, uranus, neptune.
Write a function `planets::place(const std::string& name)` that returns the place of the planet in that order, counting from 1, or 0 when the name is not in the list.
Limits: 0 ≤ length of the name ≤ 20; the name is in lower case.
## Solution
```cpp
#pragma once

#include <string>
#include <vector>

namespace planets {

//: Make a table of fixed values.
const std::vector<std::string> order {
    [["mercury"]], "venus", "earth", "mars",
    "jupiter", "saturn", "uranus", "neptune"};

//: Write a function that gives back a value worked out in one step.
[[std::vector<std::string>]] names()
{
    //:: Give back the table of planets.
    return [[order]];
}

//: Write a function that finds the first item that passes a test and gives back its position.
int place([[const std::string& name]])
{
    //:: Give back the place of the first planet equal to the name, counting from 1.
    for (int index {0}; index < std::ssize(order); ++index) {
        if ([[order[index] == name]]) {
            return index + 1;
        }
    }
    //:: Give back 0 when no planet is equal to it.
    return [[0]];
}

}  // namespace planets
```
## Slow
```cpp
#pragma once

#include <string>
#include <vector>

namespace planets {

inline std::vector<std::string> names()
{
    std::vector<std::string> all {};
    all.push_back("mercury");
    all.push_back("venus");
    all.push_back("earth");
    all.push_back("mars");
    all.push_back("jupiter");
    all.push_back("saturn");
    all.push_back("uranus");
    all.push_back("neptune");
    return all;
}

inline int place(const std::string& name)
{
    int found {0};
    const std::vector<std::string> all {names()};
    for (int index {std::ssize(all) - 1}; index >= 0; --index) {
        if (all[index] == name) {
            found = index + 1;
        }
    }
    return found;
}

}  // namespace planets
```

    REQUIRE(planets::names() == expected);
}
TEST_CASE("each_place_matches_the_list") {
    const std::vector<std::string> all {planets::names()};
