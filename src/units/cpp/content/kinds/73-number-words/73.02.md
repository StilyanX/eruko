# 73.02
Source: written
Write a function `short_scale::say(long long number)` that returns the number as a `std::string` of its groups of 3 digits, counted from the right, each written in digits with no zeros in front and followed by its word.
The words, from the right, are none, `thousand`, `million` and `billion`; a group of 0 is left out, and a single space stands between 2 parts.
1234567 gives `1 million 234 thousand 567`, 2000500 gives `2 million 500`, and 0 gives `0`.
Limits: 0 ≤ number ≤ 999 999 999 999.
## Solution
```cpp
#pragma once

#include <string>

namespace short_scale {

//: Write a function that tests each case in order and gives back the result of the first that holds.
std::string power_word([[int power]])
{
    //:: Give back no word for power 0, `thousand` for 1, `million` for 2, else `billion`.
    if ([[power == 0]]) {
        return "";
    } else if (power == 1) {
        return " thousand";
    } else if (power == 2) {
        return " million";
    } else {
        return " billion";
    }
}

//: Write a function that tests each case in order and gives back the result of the first that holds.
std::string group_words([[long long group]], int power)
{
    //:: Give back nothing for a group of 0, else the group and the word for its power.
    if ([[group == 0]]) {
        return "";
    } else {
        return std::to_string(group) + power_word(power);
    }
}

//: Write a function that repeats a step until a condition holds and gives back the result.
std::string say([[long long number]])
{
    //:: Give back `0` for 0.
    if ([[number == 0]]) {
        return "0";
    }
    //:: Start the result as an empty string.
    std::string result {[[""]]};
    //:: Start the power at 0.
    int power {[[0]]};
    //:: While the number is above 0, put the words for its last 3 digits, when not 0, at the front of the result, then drop them.
    while ([[number > 0]]) {
        std::string words {group_words(number % 1000, power)};
        if (!words.empty() && !result.empty()) {
            words += " " + result;
        }
        if (!words.empty()) {
            result = words;
        }
        number /= 1000;
        ++power;
    }
    //:: Give back the result.
    return [[result]];
}

}  // namespace short_scale
```
## Slow
```cpp
#pragma once

#include <string>
#include <vector>

namespace short_scale {

std::string say(long long number)
{
    const std::string digits {std::to_string(number)};
    const std::vector<std::string> words {
        "", " thousand", " million", " billion"};
    std::string result {""};
    std::size_t start {0};
    std::size_t length {digits.size() % 3 == 0 ? 3 : digits.size() % 3};
    while (start < digits.size()) {
        const std::size_t left {(digits.size() - start - length) / 3};
        std::string group {digits.substr(start, length)};
        while (group.size() > 1 && group[0] == '0') {
            group.erase(0, 1);
        }
        if (group != "0" || digits.size() == 1) {
            if (!result.empty()) {
                result += " ";
            }
            result += group + words[left];
        }
        start += length;
        length = 3;
    }
    return result;
}

}  // namespace short_scale
```

TEST_CASE("millions") { REQUIRE(short_scale::say(1234567) == "1 million 234 thousand 567"); }
TEST_CASE("zero") { REQUIRE(short_scale::say(0) == "0"); }
TEST_CASE("one_digit") { REQUIRE(short_scale::say(7) == "7"); }
TEST_CASE("three_digits") { REQUIRE(short_scale::say(999) == "999"); }
TEST_CASE("one_thousand") { REQUIRE(short_scale::say(1000) == "1 thousand"); }
TEST_CASE("zeros_inside_dropped")
{
    REQUIRE(short_scale::say(1000001) == "1 million 1");
}

{
    REQUIRE(short_scale::say(2000500) == "2 million 500");
}

{
    REQUIRE(short_scale::say(5007) == "5 thousand 7");

{
    REQUIRE(short_scale::say(999999999999)
