# 15.04
Source: exercism meetup
Write a class `meetup::scheduler` whose objects each hold a month of a year, as a `boost::gregorian::date::month_type` and a `boost::gregorian::date::year_type`.
Its constructor takes the month, then the year.
`month()` returns the month.
`first_monday()` through `fourth_sunday()` return the date of the first, second, third or fourth such weekday of the month, and `last_monday()` through `last_sunday()` the last one, each a `boost::gregorian::date`.
`monteenth()` through `sunteenth()` return the date of that weekday that falls on the 13th to the 19th.
Limits: none
## Solution
```cpp
#pragma once

#include <boost/date_time/gregorian/gregorian.hpp>

namespace meetup {

//: Write a class that keeps its data in private members.
class scheduler {
    using date = boost::gregorian::date;
    using days = boost::gregorian::days;
    using weekdays = boost::date_time::weekdays;
    date::month_type [[month_]];
    date::year_type year_;
public:
    //: Write a constructor that sets the data.
    scheduler(date::month_type month, date::year_type year)
        : month_ {month}, year_ {[[year]]}
    {
    }
    //: Write a member function that gives back the data.
    date::month_type month() const
    {
        return [[month_]];
    }
    //: Write member functions that give back a value worked out in one step.
    date from(weekdays day, int start, int step) const
    {
        date found {year_, month_, start};
        while ([[found.day_of_week().as_enum() != day]]) {
            found += days {step};
        }
        return found;
    }
    date nth(weekdays day, int n) const
    {
        return from(day, 1, 1) + days {7 * (n - 1)};
    }
    date last(weekdays day) const
    {
        const date first {year_, month_, 1};
        return from(day, first.end_of_month().day(), -1);
    }
    date teenth(weekdays day) const
    {
        return from(day, 13, 1);
    }
    date monteenth() const
    {
        return teenth(weekdays::Monday);
    }
    date tuesteenth() const
    {
        return teenth(weekdays::Tuesday);
    }
    date wednesteenth() const
    {
        return teenth(weekdays::Wednesday);
    }
    date thursteenth() const
    {
        return teenth(weekdays::Thursday);
    }
    date friteenth() const
    {
        return teenth(weekdays::Friday);
    }
    date saturteenth() const
    {
        return teenth(weekdays::Saturday);
    }
    date sunteenth() const
    {
        return teenth(weekdays::Sunday);
    }
    date first_monday() const
    {
        return nth(weekdays::Monday, 1);
    }
    date first_tuesday() const
    {
        return nth(weekdays::Tuesday, 1);
    }
    date first_wednesday() const
    {
        return nth(weekdays::Wednesday, 1);
    }
    date first_thursday() const
    {
        return nth(weekdays::Thursday, 1);
    }
    date first_friday() const
    {
        return nth(weekdays::Friday, 1);
    }
    date first_saturday() const
    {
        return nth(weekdays::Saturday, 1);
    }
    date first_sunday() const
    {
        return nth(weekdays::Sunday, 1);
    }
    date second_monday() const
    {
        return nth(weekdays::Monday, 2);
    }
    date second_tuesday() const
    {
        return nth(weekdays::Tuesday, 2);
    }
    date second_wednesday() const
    {
        return nth(weekdays::Wednesday, 2);
    }
    date second_thursday() const
    {
        return nth(weekdays::Thursday, 2);
    }
    date second_friday() const
    {
        return nth(weekdays::Friday, 2);
    }
    date second_saturday() const
    {
        return nth(weekdays::Saturday, 2);
    }
    date second_sunday() const
    {
        return nth(weekdays::Sunday, 2);
    }
    date third_monday() const
    {
        return nth(weekdays::Monday, 3);
    }
    date third_tuesday() const
    {
        return nth(weekdays::Tuesday, 3);
    }
    date third_wednesday() const
    {
        return nth(weekdays::Wednesday, 3);
    }
    date third_thursday() const
    {
        return nth(weekdays::Thursday, 3);
    }
    date third_friday() const
    {
        return nth(weekdays::Friday, 3);
    }
    date third_saturday() const
    {
        return nth(weekdays::Saturday, 3);
    }
    date third_sunday() const
    {
        return nth(weekdays::Sunday, 3);
    }
    date fourth_monday() const
    {
        return nth(weekdays::Monday, 4);
    }
    date fourth_tuesday() const
    {
        return nth(weekdays::Tuesday, 4);
    }
    date fourth_wednesday() const
    {
        return nth(weekdays::Wednesday, 4);
    }
    date fourth_thursday() const
    {
        return nth(weekdays::Thursday, 4);
    }
    date fourth_friday() const
    {
        return nth(weekdays::Friday, 4);
    }
    date fourth_saturday() const
    {
        return nth(weekdays::Saturday, 4);
    }
    date fourth_sunday() const
    {
        return nth(weekdays::Sunday, 4);
    }
    date last_monday() const
    {
        return last(weekdays::Monday);
    }
    date last_tuesday() const
    {
        return last(weekdays::Tuesday);
    }
    date last_wednesday() const
    {
        return last(weekdays::Wednesday);
    }
    date last_thursday() const
    {
        return last(weekdays::Thursday);
    }
    date last_friday() const
    {
        return last(weekdays::Friday);
    }
    date last_saturday() const
    {
        return last(weekdays::Saturday);
    }
    date last_sunday() const
    {
        return last(weekdays::Sunday);
    }
};

}  // namespace meetup
```
