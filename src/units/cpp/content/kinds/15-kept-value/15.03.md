# 15.03
Source: exercism robot-name
Write a class `robot_name::robot` whose objects each hold a name: 2 capital letters, then 3 digits (`RX837`), picked at random.
Its constructor gives the robot a name that no robot has had before.
`name()` returns the name.
`reset()` gives the robot a new name that no robot has had before.
Limits: none
## Solution
```cpp
#pragma once

#include <random>
#include <string>
#include <unordered_set>

namespace robot_name {

//: Write a class that keeps its data in private members.
class robot {
    std::string name_ [[{}]];
    inline static std::unordered_set<std::string> used_ {};
    inline static std::mt19937 engine_ {
        std::random_device {}()};
public:
    //: Write a constructor that sets the data.
    robot()
        : name_ {[[fresh()]]}
    {
    }
    void reset()
    {
        name_ = fresh();
    }
    //: Write a member function that gives back the data.
    const std::string& name() const
    {
        return [[name_]];
    }
    //: Write member functions that give back a value worked out in one step.
    static std::string fresh()
    {
        std::uniform_int_distribution<int> letter {'A', 'Z'};
        std::uniform_int_distribution<int> digit {'0', '9'};
        std::string name {};
        do {
            name = "";
            for (int i {0}; i < 5; ++i) {
                name += static_cast<char>(
                    i < 2 ? letter(engine_) : digit(engine_));
            }
        } while ([[!used_.insert(name).second]]);
        return name;
    }
};

}  // namespace robot_name
```
