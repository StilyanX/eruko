# 15.02
Source: written
Write a class `square::square` whose constructor takes the length of a side as an `int`.
`side()` returns that length.
`perimeter()` returns 4 times the side, and `area()` returns the side times itself.
Limits: 0 ≤ side ≤ 1000.
## Solution
```cpp
#pragma once

namespace square {

//: Write a class that keeps its data in private members.
class square {
    int side_ [[{0}]];
public:
    //: Write a constructor that sets the data.
    explicit square(int side)
        : side_ {[[side]]}
    {
    }
    //: Write a member function that gives back the data.
    int side() const
    {
        return [[side_]];
    }
    //: Write member functions that give back a value worked out in one step.
    int perimeter() const
    {
        return [[4 * side_]];
    }
    int area() const
    {
        return side_ * side_;
    }
};

}  // namespace square
```
## Slow
```cpp
#pragma once

namespace square {

class square {
    int side_ {0};
public:
    explicit square(int side)
        : side_ {side}
    {
    }
    int side() const
    {
        return side_;
    }
    int perimeter() const
    {
        int total {0};
        for (int edge {0}; edge < 4; ++edge) {
            total += side_;
        }
        return total;
    }
    int area() const
    {
        int total {0};
        for (int row {0}; row < side_; ++row) {
            for (int column {0}; column < side_; ++column) {
                ++total;
            }
        }
        return total;
    }
};

}  // namespace square
```
