# 15.01
Source: exercism space-age
Write a class `space_age::space_age` whose constructor takes an age in seconds as an `unsigned long long`.
`seconds()` returns that number of seconds.
`on_earth()` returns the age as a `double` in Earth years, where one Earth year is 31 557 600 seconds.
`on_mercury()`, `on_venus()`, `on_mars()`, `on_jupiter()`, `on_saturn()`, `on_uranus()` and `on_neptune()` each return the age as a `double` in years of that planet.
A planet's year, in Earth years, is: Mercury 0.2408467, Venus 0.61519726, Mars 1.8808158, Jupiter 11.862615, Saturn 29.447498, Uranus 84.016846, Neptune 164.79132.
An age in years must be right to within 0.005.
Limits: 0 ≤ seconds ≤ 10 000 000 000.
## Solution
```cpp
#pragma once

namespace space_age {

//: Write a class that keeps its data in private members.
class space_age {
    unsigned long long seconds_ [[{0}]];
public:
    //: Write a constructor that sets the data.
    explicit space_age(unsigned long long seconds)
        : seconds_ {[[seconds]]}
    {
    }
    //: Write a member function that gives back the data.
    unsigned long long seconds() const
    {
        return [[seconds_]];
    }
    //: Write member functions that give back a value worked out in one step.
    double on_earth() const
    {
        return [[seconds_ / 31557600.0]];
    }
    //:: Divide the age in Earth years by the planet's year.
    double on_mercury() const
    {
        return [[on_earth() / 0.2408467]];
    }
    double on_venus() const
    {
        return on_earth() / 0.61519726;
    }
    double on_mars() const
    {
        return on_earth() / 1.8808158;
    }
    double on_jupiter() const
    {
        return on_earth() / 11.862615;
    }
    double on_saturn() const
    {
        return on_earth() / 29.447498;
    }
    double on_uranus() const
    {
        return on_earth() / 84.016846;
    }
    double on_neptune() const
    {
        return on_earth() / 164.79132;
    }
};

}  // namespace space_age
```
