# 57.02
Source: written
Write a class `distinct::picker` whose constructor takes a length as an `int`.
Its member `pick(const std::vector<std::string>& words)` returns a `std::vector<std::string>` of the words, in the order given, that have exactly that many letters and in which no letter comes twice.
Upper and lower case count as the same letter, so `Anna` has `a` twice, and each word is returned as given.
Limits: 1 ≤ length ≤ 10; 0 ≤ number of words ≤ 20; each word 1 to 10 letters `A` to `Z` or `a` to `z`.
## Solution
```cpp
#pragma once

#include <algorithm>
#include <cctype>
#include <string>
#include <vector>

namespace distinct {

//: Write a function that gives back a copy of the text with each letter in lower case.
std::string lower([[std::string text]])
{
    for (char& character : text) {
        character = static_cast<char>(std::tolower(character));
    }
    return text;
}

//: Write a function that gives back a copy of the text with its characters in order.
std::string sorted([[std::string text]])
{
    std::ranges::sort(text);
    return text;
}

//: Write a class that keeps its data in private members.
class picker {
    int length_ [[{0}]];
public:
    //: Write a constructor that sets the data.
    explicit picker(int length)
        : length_ {[[length]]}
    {
    }
    //: Write a member function that gives back the items that pass a test.
    std::vector<std::string> pick(
        [[const std::vector<std::string>& words]]) const
    {
        //:: Start the result as an empty list.
        std::vector<std::string> result [[{}]];
        //:: Add each word of the length in which no letter comes twice.
        for (const std::string& word : words) {
            const std::string letters {sorted(lower(word))};
            const bool once {std::ranges::adjacent_find(letters)
                             == letters.end()};
            if ([[once && std::ssize(word) == length_]]) {
                result.push_back(word);
            }
        }
        //:: Give back the result.
        return [[result]];
    }
};

}  // namespace distinct
```
## Slow
```cpp
#pragma once

#include <cctype>
#include <string>
#include <vector>

namespace distinct {

class picker {
    int length_ {0};
public:
    explicit picker(int length)
        : length_ {length}
    {
    }
    std::vector<std::string> pick(
        const std::vector<std::string>& words) const
    {
        std::vector<std::string> result {};
        for (const std::string& word : words) {
            int repeats {0};
            for (std::size_t i {0}; i < word.size(); ++i) {
                for (std::size_t j {0}; j < word.size(); ++j) {
                    if (i != j && std::tolower(word[i])
                                      == std::tolower(word[j])) {
                        ++repeats;
                    }
                }
            }
            if (static_cast<int>(word.size()) == length_
                && repeats == 0) {
                result.push_back(word);
            }
        }
        return result;
    }
};

}  // namespace distinct
```
## Tests
```cpp
TEST_CASE("four_letters") {
    const distinct::picker four {4};
    const std::vector<std::string> words {"word", "noon", "cat", "Lamp"};
    const std::vector<std::string> expected {"word", "Lamp"};
    REQUIRE(four.pick(words) == expected);
}
TEST_CASE("no_words") {
    const distinct::picker three {3};
    REQUIRE(three.pick({}).empty());
}
TEST_CASE("case_does_not_matter") {
    const distinct::picker four {4};
    const std::vector<std::string> words {"Anna", "ABcD", "aBcA"};
    const std::vector<std::string> expected {"ABcD"};
    REQUIRE(four.pick(words) == expected);
}
TEST_CASE("one_letter") {
    const distinct::picker one {1};
    const std::vector<std::string> words {"a", "B", "it"};
    const std::vector<std::string> expected {"a", "B"};
    REQUIRE(one.pick(words) == expected);
}
TEST_CASE("ten_letters") {
    const distinct::picker ten {10};
    const std::vector<std::string> words {
        "Background", "Pathfinder", "abcdefghij", "abcdefghia"};
    const std::vector<std::string> expected {
        "Background", "Pathfinder", "abcdefghij"};
    REQUIRE(ten.pick(words) == expected);
}
TEST_CASE("none_pass") {
    const distinct::picker five {5};
    const std::vector<std::string> words {"apple", "tree", "letters"};
    REQUIRE(five.pick(words).empty());
}
TEST_CASE("order_is_kept") {
    const distinct::picker three {3};
    const std::vector<std::string> words {
        "zoo", "zip", "Ape", "bee", "cow", "Dad"};
    const std::vector<std::string> expected {"zip", "Ape", "cow"};
    REQUIRE(three.pick(words) == expected);
}
```
