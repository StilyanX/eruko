# 55.02
Source: written
Write a class `hex_color::color` whose constructor takes a `const std::string& text` and keeps the letters and digits of text, in order and made small, dropping every other character.
It throws `std::domain_error` unless 6 are left and each is a digit or a letter from `a` to `f`.
`code()` returns the 6 kept as a `std::string`.
Limits: 0 ≤ length of the text ≤ 100.
## Solution
```cpp
#pragma once

#include <cctype>
#include <stdexcept>
#include <string>

namespace hex_color {

//: Write a class that keeps its data in private members.
class color {
    [[std::string]] code_ {};
public:
    //: Write a constructor that sets the data, and stops with an error when it is bad.
    explicit color([[const std::string& text]])
    {
        //:: Keep each letter and digit of the text, made small.
        for (const char character : text) {
            if ([[std::isalnum(character)]]) {
                code_ += static_cast<char>(
                    std::tolower(character));
            }
        }
        //:: Stop with an error unless 6 are left, each a digit or a letter from `a` to `f`.
        if ([[code_.size() != 6]]) {
            throw std::domain_error {"not a color"};
        }
        for (const char digit : code_) {
            if (!std::isxdigit(digit)) {
                throw std::domain_error {"not a color"};
            }
        }
    }
    //: Write a member function that gives back the data.
    std::string code() const
    {
        return [[code_]];
    }
};

}  // namespace hex_color
```
## Slow
```cpp
#pragma once

#include <stdexcept>
#include <string>

namespace hex_color {

class color {
    std::string code_ {};
public:
    explicit color(const std::string& text)
    {
        const std::string small {"abcdefghijklmnopqrstuvwxyz"};
        const std::string large {"ABCDEFGHIJKLMNOPQRSTUVWXYZ"};
        const std::string digits {"0123456789"};
        for (const char character : text) {
            for (std::size_t index {0}; index < 26; ++index) {
                if (character == small[index]
                    || character == large[index]) {
                    code_ += small[index];
                }
            }
            for (const char digit : digits) {
                if (character == digit) {
                    code_ += digit;
                }
            }
        }
        bool good {code_.size() == 6};
        for (const char kept : code_) {
            if (kept > 'f') {
                good = false;
            }
        }
        if (!good) {
            throw std::domain_error {"not a color"};
        }
    }
    std::string code() const
    {
        return code_;
    }
};

}  // namespace hex_color
```

    REQUIRE(hex_color::color("000000").code() == "000000");
